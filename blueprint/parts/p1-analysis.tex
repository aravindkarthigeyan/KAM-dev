Throughout, functions on strips are holomorphic on the open strip and bounded there, and norms are
sup norms (Part~\ref{bp:p0}). Constants called $c(\cdot)$ depend only on the displayed quantities.

\subsection{Cauchy estimates}

\begin{lemma}[Cauchy estimate on strips]\label{bp:p1:cauchy-strip}
Let $u$ be holomorphic and bounded on $T_\rho$ and $0<\delta<\rho$. Then for every $\ell$,
$\norm{\partial_{\theta_\ell}u}_{\rho-\delta}\le\delta^{-1}\norm{u}_\rho$, hence
$\norm{Du}_{\rho-\delta}\le n\delta^{-1}\norm{u}_\rho$; the same holds for vector- and matrix-valued
$u$ with the norms of Part~\ref{bp:p0}.
\source{HL19 \S3.1, p.~15.}
\end{lemma}

\begin{lemma}[Cauchy estimate on open sets]\label{bp:p1:cauchy-open}
Let $f$ be holomorphic on an open $U\subset\C^m$ with $\abs f\le A$ on $U$, and let the closed
polydisc of radius $\delta$ around $z$ lie in $U$. Then $\abs{\partial_\ell f(z)}\le A/\delta$, and
$\abs{D^jf(z)}\le c(m,j)A\delta^{-j}$ for $j=1,2,3$, where $D^j$ is the iterated Fréchet derivative and
$\abs{\cdot}$ its operator norm (Lean \Lean{iteratedFDeriv}).
\end{lemma}

\subsection{Fourier coefficients of periodic holomorphic functions}

\begin{lemma}[Coefficient decay]\label{bp:p1:decay}
Let $u$ be holomorphic, bounded and $1$-periodic on $T_\rho$. Then
$\abs{\hat u_k}\le e^{-2\pi\abs{k}_1\rho'}\norm{u}_\rho$ for every $k\in\Z^n$ and $0\le\rho'<\rho$.
\end{lemma}

\begin{lemma}[Summation]\label{bp:p1:summation}
If $\sum_k\abs{a_k}e^{2\pi\abs{k}_1\rho}<\infty$, then $u(\theta)=\sum_ka_ke^{2\pi ik\cdot\theta}$ is
holomorphic and $1$-periodic on $T_\rho$, $\hat u_k=a_k$, and
$\norm{u}_\rho\le\sum_k\abs{a_k}e^{2\pi\abs{k}_1\rho}$. Conversely, for $u$ as in
Lemma~\ref{bp:p1:decay} and $0<\delta<\rho$,
$\sum_k\abs{\hat u_k}e^{2\pi\abs{k}_1(\rho-\delta)}\le\coth(\pi\delta)^n\norm{u}_\rho$, and $u$ equals
its Fourier series on $T_{\rho}$.
\end{lemma}

\subsection{The cohomological equation}

\begin{lemma}[Cohomological equation on strips]\label{bp:p1:cohomological}
Let $\omega\in\Dio_{\gamma,\tau}$ with $\gamma>0$, $\tau\ge0$, and let $g$ be holomorphic, bounded
and $1$-periodic on $T_\rho$ with $\avg g=0$. For $0<\delta<\rho$ there is a unique $u$ in the class of functions
holomorphic and $1$-periodic on $T_{\rho-\delta}$ (no boundedness assumed), with $\avg u=0$ and
$Du(\theta)\omega=g(\theta)$; this $u$ is bounded, and
\[\norm{u}_{\rho-\delta}\le c(n,\tau)\,\gamma^{-1}\delta^{-(\tau+n)}\norm{g}_\rho.\]
If $g$ is real on real points, so is $u$. The same holds componentwise for vector- and
matrix-valued $g$.
\source{HL19 \S4.1 (with Rüssmann's sharper exponent $\tau$; the cruder exponent $\tau+n$ suffices,
decision D12). Coefficient form: \texttt{docs/cohomological-equation.md}, Lean \Lean{KAM.Diophantine}.}
\end{lemma}

\begin{lemma}[Averages]\label{bp:p1:averages}
For $v$ holomorphic, bounded and $1$-periodic on $T_\rho$: $\avg{\partial_{\theta_\ell}v}=0$,
$\avg{Dv\,\omega}=0$, and $\abs{\avg v}\le\norm{v}_\rho$.
\end{lemma}

\subsection{Matrices}

\begin{lemma}[Neumann series]\label{bp:p1:neumann}
If $A$ is invertible and $\abs{A^{-1}}\abs{\Delta}\le\frac12$, then $A+\Delta$ is invertible with
$\abs{(A+\Delta)^{-1}}\le2\abs{A^{-1}}$ and $\abs{(A+\Delta)^{-1}-A^{-1}}\le2\abs{A^{-1}}^2\abs{\Delta}$.
The same holds pointwise for matrix-valued functions, with sup norms.
\end{lemma}

\subsection{Further interface lemmas (transferred from Part 3, 27 Oct 2026)}

\begin{lemma}[Periodic functions are bounded on smaller strips]\label{bp:p1:periodic-bounded}
If $u$ is holomorphic and $1$-periodic on $T_\rho$, then $u$ is bounded on $T_{\rho'}$ for every $\rho'<\rho$.
\end{lemma}

\begin{lemma}[Increments]\label{bp:p1:increments}
If $u$ is holomorphic on $T_\rho$ and $\theta,\theta+v\in T_{\rho-\delta}$ with the segment between them in
$T_{\rho-\delta}$, then $\abs{u(\theta+v)-u(\theta)}\le\norm{Du}_{\rho-\delta}\abs{v}$; with
Lemma~\ref{bp:p1:cauchy-strip}, $\le n\delta^{-1}\norm{u}_\rho\abs{v}$.
\end{lemma}

\begin{lemma}[Second-order Taylor bound]\label{bp:p1:taylor}
If $F$ is holomorphic on an open $U\subset\C^m$ with $\abs{D^2F}\le A$ on $U$ and the segment $[z,z+w]\subset U$, then
$\abs{F(z+w)-F(z)-DF(z)w}\le\frac A2\abs{w}^2$.
\end{lemma}

\begin{lemma}[Max-norm matrix facts]\label{bp:p1:matrix-norms}
For the norm $\abs{M}=\max_i\sum_j\abs{M_{ij}}$: $\abs{MN}\le\abs M\abs N$, $\abs{Mv}\le\abs M\abs v$, and
$\abs{M^\top}\le m\abs M$ for $m\times m'$ matrices with $m'\le m$ (dimension factors explicit).
\end{lemma}

\begin{lemma}[Holomorphic inverse]\label{bp:p1:holomorphic-inverse}
If $A$ is a holomorphic matrix-valued function on an open set with $\det A\ne0$ there, then $A^{-1}$ is
holomorphic there, and $D(A^{-1})=-A^{-1}(DA)A^{-1}$.
\end{lemma}

\begin{lemma}[Quadratic iteration on nested strips]\label{bp:p1:iteration}
Let $Q>0$, $p\ge0$ and $y_0,y_1,\dots\ge0$ with $y_{m+1}\le Q2^{pm}y_m^2$ for every $m\ge0$ and
$Q2^py_0\le\frac12$. Then $y_m\le Q^{-1}2^{-p(m+1)}2^{-2^m}$ for every $m\ge0$; in particular $y_m\to0$.
If moreover only $q\coloneqq Q2^py_0\le1$ is assumed, then $y_m\le q^{2^m-1}2^{-pm}y_0$ for every $m\ge0$
(linear in $y_0$, as needed for closeness bounds). (Proved as \texttt{bp:p3:iteration} and
\texttt{bp:p2:error-decay}(a).)
\end{lemma}

\begin{lemma}[Averages of functions on the real torus]\label{bp:p1:averages-real}
For $v:\R^n\to\C$ continuous, $1$-periodic and differentiable with continuous derivative:
$\avg{\partial_{\theta_\ell}v}=0$ and $\avg{Dv\,\omega}=0$.
\end{lemma}

\subsection{Further interface lemmas (transferred from Part 2, 27 Oct 2026)}

\begin{lemma}[Operator norms, blocks, $\Omega_0$]\label{bp:p1:operator-norms}
For the max norms of Part~\ref{bp:p0}: the operator norm of a matrix is $\max_i\sum_j\abs{M_{ij}}$; for a block
matrix $\left(\begin{smallmatrix}A&B\\C&D\end{smallmatrix}\right)$ the norm is at most $\max\{\abs A+\abs B,\abs C+\abs D\}$;
$\abs{\Omega_0}=1$, $\Omega_0^\top=-\Omega_0$, $\Omega_0^2=-I$. (Proved as \texttt{bp:p2:matrix}(a), (c), (d).)
\end{lemma}

\begin{lemma}[Cauchy corollaries]\label{bp:p1:cauchy-corollaries}
For $f$ holomorphic and bounded on $T_\rho$ (scalar, vector or matrix valued) and $0<\delta<\rho$:
$\norm{(Df)^\top}_{\rho-\delta}\le c\,\delta^{-1}\norm f_\rho$ and
$\norm{\partial_\omega f}_{\rho-\delta}\le\abs\omega_1\delta^{-1}\norm f_\rho$, with $c$ a dimension factor.
(Proved as \texttt{bp:p2:cauchy}.)
\end{lemma}

\begin{lemma}[Averages of matrix-valued functions]\label{bp:p1:averages-matrix}
For $M$ continuous, $1$-periodic and matrix valued on $\R^n$: $\abs{\avg M}\le\sup_{\R^n}\abs M$, and $\avg{\partial_\omega M}=0$ if
$M$ is $C^1$. (Proved as \texttt{bp:p2:averages}.)
\end{lemma}

\begin{lemma}[Cohomological equation in operator norms, a-priori bound]\label{bp:p1:cohomological-operator}
Lemma~\ref{bp:p1:cohomological} holds for matrix-valued $g$ with the operator norm, and every holomorphic periodic
solution $u$ with $\avg u=0$ satisfies the same bound. (Proved as \texttt{bp:p2:cohomological}.)
\end{lemma}

\begin{lemma}[Increments on open sets]\label{bp:p1:increments-open}
If $F$ is holomorphic on an open $U\subset\C^m$, $\abs{DF}\le A$ on $U$, and the segment $[z,z+w]\subset U$, then
$\abs{F(z+w)-F(z)}\le A\abs w$.
\end{lemma}

\begin{lemma}[Real inverses]\label{bp:p1:real-inverse}
The inverse of an invertible real matrix is real; hence if $A(\theta)$ is real for real $\theta$ and invertible, so is
$A(\theta)^{-1}$.
\end{lemma}

\begin{lemma}[Uniform limits]\label{bp:p1:uniform-limits}
If $K_s$ are holomorphic on an open set and $K_s\to K$, $DK_s\to\Lambda$ uniformly there, then $K$ is holomorphic and
$DK=\Lambda$; primary periodicity and realness on reals pass to the limit.
\end{lemma}

\subsection{Further interface lemmas (transferred from Part 4, 27 Oct 2026)}

\begin{lemma}[Second-derivative Cauchy estimate on strips]\label{bp:p1:cauchy-strip-2}
For $u$ holomorphic and bounded on $T_\rho$ and $0<\delta<\rho$: $\norm{D^2u}_{\rho-\delta}\le c(n)\delta^{-2}\norm u_\rho$.
\end{lemma}

\begin{lemma}[Translation invariance of averages]\label{bp:p1:periodic-average}
For $v$ continuous and $1$-periodic on $\R^n$ and $a\in\R^n$: $\avg{v(\cdot+a)}=\avg v$. (Proved as \texttt{bp:p4:periodic-average}.)
\end{lemma}

\begin{lemma}[Real restriction]\label{bp:p1:real-restriction}
If $K$ is holomorphic on $T_\rho$ and real on reals, its restriction to $\R^n$ is real-analytic, and its real derivative is
the restriction of its complex derivative. (Proved as \texttt{bp:p4:real-restriction}.)
\end{lemma}

\begin{lemma}[Real and complex Hamiltonian fields]\label{bp:p1:hamiltonian-real}
If $h$ is holomorphic near a real point $z$ and real on real points, then $X_{\operatorname{Re}h}(z)=X_h(z)$, and
$X_\varphi=J\,D\varphi^\top$ in the frozen conventions. (Proved as \texttt{bp:p4:hamiltonian-real}.)
\end{lemma}

\begin{lemma}[Mixed second differences]\label{bp:p1:mixed-difference}
For $F$ holomorphic on a convex open set with $\abs{D^2F}\le A$:
$\abs{F(z+v+w)-F(z+v)-F(z+w)+F(z)}\le A\abs v\abs w$. (Proved as \texttt{bp:p4:mixed-difference}.)
\end{lemma}

\begin{lemma}[Lipschitz images and preimages]\label{bp:p1:lip-image}
A $\Lambda$-Lipschitz map on $\R^d$ (max norm) multiplies outer Lebesgue measure by at most $\Lambda^d$; the preimage of a
null set under a map with Lipschitz inverse is null. (Proved as \texttt{bp:p4:lip-image}, \texttt{bp:p4:preimage-measure}.)
\end{lemma}

\begin{lemma}[McShane extension]\label{bp:p1:mcshane}
A $\Lambda$-Lipschitz map from a subset of a metric space to $\R^d$ with the max norm extends to a $\Lambda$-Lipschitz map on
the whole space; it can be clamped to keep a given sup bound. (Mathlib \Lean{LipschitzOnWith.extend\_pi}; proved as
\texttt{bp:p4:mcshane}.)
\end{lemma}

\begin{lemma}[Lipeomorphisms]\label{bp:p1:lipeo}
If $G:\R^d\to\R^d$ is $\lambda$-Lipschitz with $\lambda<1$, then $\mathrm{id}+G$ is a bi-Lipschitz bijection of $\R^d$.
(Proved as \texttt{bp:p4:lipeo}.)
\end{lemma}
