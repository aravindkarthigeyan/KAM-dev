\subsection{Real extended systems, conjugation, stopped transfer}\label{bpN:p11:s-real}

Setting of \S\ref{bpN:p11:s-ext-class}: $I_c$, $S$, $r_B,s,w$, $\lambda,c$, $\tilde S$ ($\lambda,c$ fixed for the block). For $0<\varrho\le r_B$ let
$W^e_\varrho=\{(\theta,I,\tilde x)\in\R^{2n+2k}:\abs{I-I_c}<\varrho,\ \operatorname{dist}_\infty(\tilde x,\tilde S)<\varrho\}=\Dom\varrho s^e\cap\R^{2n+2k}$ (open). For $G\in\Os rs^{e,\R}$ the real Hamiltonian $G^\R$ is $G|_{W^e_r}$ (and $0$ off it). A \emph{$G$-orbit in $W^e_\varrho$ on an interval $J$} is a $y:\R\to\R^{2n+2k}$ with $y(t)\in W^e_\varrho$ and $y'(t)=X^c_{G^\R}(y(t))$ for $t\in J$ ($y$ is defined on $\R$ and differentiable \emph{two-sidedly} at every $t\in J$, also at endpoints, exactly as in Part~\ref{bpN:p3} and the frozen \texttt{IsSolutionOn}; all helper statements below use this predicate, which the frozen input satisfies).

\begin{lemma}[Real extended systems]\label{bpN:p11:real-system}
Let $G$ be holomorphic and $1$-periodic in $\theta$ on $\Dom rs^e$, real on $W^e_r$ (finite norm is needed only in (c), (d)), $0<\varrho\le r$.
(a) $G^\R\in C^\infty(W^e_r)$, $G^\R$ is $1$-periodic in $\theta$, and $X^c_{G^\R}(y)=X^c_G(y)\in\R^{2n+2k}$ (complex partials at the real point $y$); $X^c_{G^\R}$ is $C^1$ and $\theta$-periodic on $W^e_r$.
(b) (Energy) If $y$ is a $G$-orbit in $W^e_\varrho$ on $J$ then $G^\R\circ y$ is constant on $J$.
(c) For $u\in\Os\varrho s^e$ with finite norm and $y\in W^e_\varrho$: $\abs{u(y)}\le\wn u\varrho s$ and $\sum_j\abs{\partial_{\theta_j}u(y)}\le\wn u\varrho s/(es)$.
(d) If $P_\Lambda g=g$ ($g\in\Os\varrho s^e$ finite norm, $\Lambda\subset\Z^n$) then $\Pi_\Lambda^\perp\partial_\theta g=0$ on $W^e_\varrho$.
\end{lemma}
\begin{proof}
\proofstep{(a)}\label{stepN:p11:rs-a} Lemma~\ref{bpN:p7:real-system}(a),(b),(c) with $\Dom rs^e\cap\R^{2n+2k}=W^e_r$ and \Kref{bp:p1:hamiltonian-real}: the proof only uses holomorphy on an open set and reality at the real points, not the structure of the variables.
\proofstep{(b)}\label{stepN:p11:rs-b} $G^\R$ is differentiable and $\frac d{dt}G^\R(y(t))=\nabla G^\R(y)\cdot\Omega_c\nabla G^\R(y)=0$ ($\Omega_c^\top=-\Omega_c$); a function with zero derivative on an interval is constant (mean value inequality, \Kref{bp:p1:ext-mvt}).
\proofstep{(c),(d)}\label{stepN:p11:rs-cd} Lemma~\ref{bpN:p7:theta-derivative} and Lemma~\ref{bpN:p7:resonant-gradient}: both are statements about Fourier coefficients in $\theta$ at fixed $\hat I$ and use Lemma~\ref{bpN:p11:ext-part4} only.
\end{proof}

\begin{lemma}[Conjugation]\label{bpN:p11:conj}
Let $H\in\Os rs^{e,\R}$, $\tilde H\in\Os{\tilde r}{\tilde s}^{e,\R}$ ($\tilde r\le r$, $\tilde s\le s$ in the same extended structure), $\Psi:\Dom{\tilde r}{\tilde s}^e\to\Dom rs^e$ holomorphic, real at real points, with $D\Psi\,\Omega_c\,D\Psi^\top=\Omega_c$ and $\tilde H=H\circ\Psi$. If $\tilde y$ is an $\tilde H$-orbit in $W^e_{\tilde r}$ on $J$ then $\Psi\circ\tilde y$ is an $H$-orbit in $W^e_r$ on $J$.
\end{lemma}
\begin{proof}
\proofstep{Chain rule}\label{stepN:p11:cj} As Lemma~\ref{bpN:p7:conjugation}: $Y=D\Psi(\tilde y)$ is real, $\nabla\tilde H^\R=Y^\top\nabla H^\R\circ\Psi$, so $\frac d{dt}\Psi(\tilde y)=Y\Omega_cY^\top\nabla H^\R(\Psi\tilde y)=X^c_{H^\R}(\Psi\tilde y)$; $\Psi$ maps $W^e_{\tilde r}$ into $W^e_r$.
\end{proof}

\begin{lemma}[Real preimages of a near-identity map]\label{bpN:p11:preimage}
Let $\Psi:\Dom{\tilde r}{\tilde s}^e\to\C^{2n+2k}$ be holomorphic, real at real points, $G=\Psi-\mathrm{id}$, $\abs{G_\theta}\le d_\theta$, $\abs{G_I}\le d_I$, $\abs{G_{\tilde x}}\le d_x$ on $\Dom{\tilde r}{\tilde s}^e$. Let $0<\varrho'\le\tilde r$, $0<\sigma'\le2\tilde s$ and $\kappa_*=\max(d_\theta/\sigma',d_I/\varrho',d_x/\varrho')\le\frac14$. Then every $y=(\theta_y,I_c,\tilde x_y)$ with $\theta_y\in\R^n$, $\tilde x_y\in\tilde S$ equals $\Psi(x)$ for some $x\in W^e_{\tilde r}$ with $\abs{\theta_x-\theta_y}\le d_\theta$, $\abs{I_x-I_c}\le d_I$, $\abs{\tilde x_x-\tilde x_y}\le d_x$.
\end{lemma}
\begin{proof}
Proof of Lemma~\ref{bpN:p7:preimage} with $\nu(a,b,\xi)=\max(\abs a/\sigma',\abs b/\varrho',\abs\xi/\varrho')$: $X=\{x\in\R^{2n+2k}:\nu(x-y)\le\kappa_*\}\subset W^e_{\tilde r}$ (since $\tilde x_y\in\tilde S$ and $\kappa_*\varrho'\le\varrho'/4<\tilde r$); $T(x)=y-G(x)$ maps $X$ to itself; the complex $\nu$-ball of radius $\frac12$ about $x\in X$ lies in $\Dom{\tilde r}{\tilde s}^e$ (imaginary parts $<\sigma'/2\le\tilde s$, $\operatorname{dist}(\tilde x',\tilde S)<\frac{\varrho'}4+\frac{\varrho'}2<\tilde r$), so Lemma~\ref{bpN:p4:general-cauchy} gives $\abs{DG}_\nu\le2\kappa_*\le\frac12$ and $T$ is a contraction on the convex set $X$ (mean value inequality); Banach's fixed point theorem (\ref{bpN:p7:ext-banach}) gives $x$.
\end{proof}


\begin{lemma}[Continuation with independent periodic and transverse coordinates]\label{bpN:p11:continuation}
Let $E=\R^n\times\R^N$ (max norm), $U\subset\R^N$ open, $W=\R^n\times U$, $F:W\to E$ of class $C^1$ with $F(\theta+\ell,v)=F(\theta,v)$ ($\ell\in\Z^n$), $x_0\in W$, $(T_-,T_+)$ and $z=(z_\theta,z_v)$ the maximal interval and solution (Theorem~\ref{bpN:p3:gen-maximal}). If $K\subset U$ is compact and $z_v(t)\in K$ for $t\in[t_*,T_+)$ for some $t_*\in[0,T_+)$, then $T_+=+\infty$.
\end{lemma}
\begin{proof}
\proofstep{Uniform data}\label{stepN:p11:ct-data} If $K=\emptyset$ the hypothesis is void ($z_v(t)\in K$ impossible); else choose $\delta\in(0,1]$ with $K_\delta=\{v:\operatorname{dist}_\infty(v,K)\le\delta\}\subset U$ (Step~\ref{stepN:p3:uc-delta}; $\delta=1$ if $U=\R^N$); $C=[-1,2]^n\times K_\delta$ is compact in $W$; let $M,L$ be the bound and Lipschitz constant of $F$ on $C$ (Lemma~\ref{bpN:p3:gen-lip}). For $x=(\theta,v)$ with $v\in K$ and $\ell=(\lfloor\theta_i\rfloor)_i$: $x'=x-(\ell,0)\in[0,1)^n\times K$, $\bar B(x',\delta)\subset C$ and $\bar B(x,\delta)=\bar B(x',\delta)+(\ell,0)\subset W$; by periodicity $F$ is bounded by $M$ and $L$-Lipschitz on $\bar B(x,\delta)$. So $\Phi=\R^n\times K$ has uniform data $(\delta,M,L)$.
\proofstep{Conclusion}\label{stepN:p11:ct-concl} $z(t)\in\Phi$ on $[t_*,T_+)$; Theorem~\ref{bpN:p3:gen-continuation} gives $T_+=\infty$.
\end{proof}

\begin{lemma}[Stopped transfer through a symplectic map]\label{bpN:p11:transfer}
Let $H,\tilde H,\Psi$ be as in Lemma~\ref{bpN:p11:conj}, $\abs\cdot_*\ge\abs\cdot$ a norm on $\R^n$, and assume
\begin{enumerate}[label=(T\arabic*)]
\item $\abs{\Pi_I\Psi(x)-I_x}_*\le d_*$ and $\abs{\Pi_{\tilde x}\Psi(x)-\tilde x_x}_\infty\le d_x$ for all $x\in W^e_{\tilde r}$, with $d_x<\tilde r$;
\item every $y=(\theta_y,I_c,\tilde x_y)$ ($\theta_y\in\R^n$, $\tilde x_y\in\tilde S$) is $\Psi(x)$ for some $x\in W^e_{\tilde r}$ with $\abs{I_x-I_c}_*\le d_*$;
\item (\emph{conditional helper; a priori confinement for slowly displaced normal-form orbits}) there are $\varrho>0$ with $d_*+\varrho<\tilde r$ and $T\in(0,\infty]$ such that every $\tilde H$-orbit $\tilde y$ in $W^e_{\tilde r}$ on an interval $\tilde J\ni0$ with $\abs{\tilde I(0)-I_c}_*\le d_*$ and $\operatorname{dist}_\infty(\tilde x(t),\tilde S)\le d_x$ for all $t\in\tilde J$ satisfies $\abs{\tilde I(t)-\tilde I(0)}_*\le\varrho$ for $t\in\tilde J$, $\abs t\le T$.
\end{enumerate}
Let $F$ be any function with $F=X^c_{H^\R}$ on $W^e_r$, $J\ni0$ an interval, $y:\R\to\R^{2n+2k}$ differentiable at every $t\in J$ (two-sided) with $y'(t)=F(y(t))$, $y(0)=(\theta_0,I_c,\tilde x_0)$, $\tilde x_0\in\tilde S$, $t_1\in J$, $\abs{t_1}\le T$, with $\tilde x(u)\in\tilde S$ for all $u$ between $0$ and $t_1$. Then $y(u)\in W^e_r$ and $\abs{I(u)-I_c}_*\le\varrho+2d_*$ for all $u$ between $0$ and $t_1$.
\source{Ours; Lemma~\ref{bpN:p7:transfer} with Corollary~\ref{bpN:p7:stopped-extension}; (T3) is a helper hypothesis of this lemma only, never a premise of the frozen theorem.}
\end{lemma}
\begin{proof}
Take $t_1>0$ (for $t_1<0$ apply it to $u\mapsto y(-u)$, $-H$, $-\tilde H$; $t_1=0$ is trivial).
\proofstep{A normal-form orbit}\label{stepN:p11:tr-orbit} By (T2) choose $x$; $\tilde y$ the maximal $\tilde H$-orbit in $W^e_{\tilde r}$ through $x$ on $(T_-,T_+)\ni0$ (Lemma~\ref{bpN:p11:real-system}(a), Theorem~\ref{bpN:p3:gen-maximal}).
\proofstep{The good set}\label{stepN:p11:tr-E} $E=\{\tau\in[0,t_1]:\ [0,\tau]\subset(T_-,T_+),\ y([0,\tau])\subset W^e_r,\ y=\Psi\circ\tilde y\text{ on }[0,\tau]\}$. $0\in E$, and \emph{$E$ is downward closed} (if $\tau\in E$ and $\tau'\le\tau$ then $\tau'\in E$ by definition). So $E$ is an interval containing $0$.
\proofstep{Bounds on every prefix}\label{stepN:p11:tr-bounds} Let $\tau\in E$ and $u\in[0,\tau]$. First the slow variables: $y(u)=\Psi(\tilde y(u))$ and (T1) give $\abs{\tilde x_{\tilde y}(u)-\tilde x(u)}\le d_x$ with $\tilde x(u)\in\tilde S$, so $\operatorname{dist}(\tilde x_{\tilde y}(u),\tilde S)\le d_x$. Then (T3) applies to $\tilde y$ on $\tilde J=[0,\tau]$: $\abs{\tilde I(u)-\tilde I(0)}_*\le\varrho$, hence $\abs{\tilde I(u)-I_c}\le\varrho+d_*<\tilde r$.
\proofstep{$E$ is open in $[0,t_1]$}\label{stepN:p11:tr-open} Let $\tau\in E$, $\tau<t_1$. $\Psi\circ\tilde y$ is an $H$-orbit in $W^e_r$ on $(T_-,T_+)$ (Lemma~\ref{bpN:p11:conj}), and $\tau<T_+$ because $[0,\tau]\subset(T_-,T_+)$. $y(\tau)=\Psi(\tilde y(\tau))\in W^e_r$ (open); by continuity $y\in W^e_r$ near $\tau$, where $F=X^c_{H^\R}$ is $C^1$; $y$ and $\Psi\circ\tilde y$ solve the same equation with the same value at $\tau$, so they agree on a neighbourhood (Theorem~\ref{bpN:p3:gen-uniqueness}). Hence $[\tau,\tau+\epsilon)\subset E$.
\proofstep{$E$ is closed}\label{stepN:p11:tr-closed} Let $\tau^*=\sup E\le t_1$. By Step~\ref{stepN:p11:tr-bounds} on every prefix $[0,\tau]$, $\tau<\tau^*$, the transverse part $v=(\tilde I,\tilde x_{\tilde y})$ of $\tilde y$ lies on $[0,\tau^*)$ in $K=\{\abs{I-\tilde I(0)}_*\le\varrho\}\times\{\tilde x:\operatorname{dist}_\infty(\tilde x,\tilde x([0,t_1]))\le d_x\}$, which is compact ($\tilde x([0,t_1])$ is a continuous image of $[0,t_1]$) and contained in the open transverse set $U=\{\abs{I-I_c}<\tilde r\}\times\{\operatorname{dist}(\tilde x,\tilde S)<\tilde r\}$ (here $\tilde x([0,t_1])\subset\tilde S$ and $d_x<\tilde r$). The field $X^c_{\tilde H^\R}$ is $C^1$ and $\theta$-periodic on $\R^n\times U\supset W^e_{\tilde r}$-part; $T_+\ge\tau^*$ (all prefixes lie in $(T_-,T_+)$), and if $T_+=\tau^*$ Lemma~\ref{bpN:p11:continuation} would give $T_+=\infty$: so $T_+>\tau^*$. By continuity $y(\tau^*)=\Psi(\tilde y(\tau^*))\in W^e_r$ and $y=\Psi\circ\tilde y$ on $[0,\tau^*]$: $\tau^*\in E$.
\proofstep{Conclusion}\label{stepN:p11:tr-concl} $E=[0,t_1]$; $I(u)-I_c=(\Pi_I\Psi(\tilde y(u))-\tilde I(u))+(\tilde I(u)-\tilde I(0))+(I_x-I_c)$, of $\abs\cdot_*$-size $\le d_*+\varrho+d_*$ by (T1), Step~\ref{stepN:p11:tr-bounds}, (T2).
\end{proof}

\subsection{Nonresonant stability (Proposition 1$^e$)}\label{bpN:p11:s-nr}

\begin{proposition}[Nonresonant stability in the extended phase space]\label{bpN:p11:nonresonant}
Let $r,s,K_a,\alpha>0$, $L\ge0$, $p>1$, $q=p/(p-1)$, $\kappa=2\pi K_as\ge6$; $r=r_B$; $h$ holomorphic on $V_r(I_c)$, real on real points; $\abs{k\cdot\omega(I_c)}\ge\alpha$ for $0<\abs k_1\le K_a$ (\ref{bpN:p6:cond-nr}, $\Lambda=\{0\}$), \ref{bpN:p6:cond-L} with $L$, $K_aLr\le\alpha/p$, (SL); $f\in\Os rs^{e,\R}$, $\wn frs\le\varepsilon$ with a number $\varepsilon>0$, $\varepsilon\le\alpha r/(2^7qK_a)$; $H=h+f$; $T_{\rm nr}=\frac{sr}{5\varepsilon}e^{\kappa/6}$. Then for every $F$ with $F=X^c_{H^\R}$ on $W^e_r$, $J\ni0$, $y$ with $y'=F(y)$ on $J$, $y(0)=(\theta_0,I_c,\tilde x_0)$, $\tilde x_0\in\tilde S$, and $t_1\in J$ with $\abs{t_1}\le T_{\rm nr}$ and $\tilde x(u)\in\tilde S$ between $0$ and $t_1$: $\abs{I(u)-I_c}\le r/2$ between $0$ and $t_1$.
\source{Pos93 p.~196 (Proposition 1), Proposition~\ref{bpN:p7:nonresonant}.}
\end{proposition}
\begin{proof}
\proofstep{Normal form}\label{stepN:p11:nr-nf} Theorem~\ref{bpN:p11:nf} ($\Lambda=\{0\}$, $g_0=\hat f_0$ is $\theta$-independent): $\Psi:\Dom{r/2}{s/6}^e\to\Dom rs^e$, $H\circ\Psi=h+g+f_*$, $\eta=\wn{f_*}{}{}\le e^{-\kappa/6}\varepsilon$, $d_I=qK_a\varepsilon/\alpha\le r/2^7$, $d_x=r/2^{10}$, $d_\theta=10q\varepsilon/(2\pi\alpha r)$.
\proofstep{(T1), (T2)}\label{stepN:p11:nr-t12} $\tilde r=r/2$, $\tilde s=s/6$, $\abs\cdot_*=\abs\cdot$, $d_*=d_I$; $d_x<\tilde r$. (T2): Lemma~\ref{bpN:p11:preimage} with $\varrho'=r/2$, $\sigma'=s/3$: $d_I/\varrho'\le\frac1{64}$, $d_x/\varrho'\le2^{-9}$, $d_\theta/\sigma'\le\frac{30}{2^7\kappa}\le\frac{30}{768}<\frac14$ (from (i), $\kappa\ge6$); the preimage has $\abs{I_x-I_c}\le d_I=d_*$.
\proofstep{(T3)}\label{stepN:p11:nr-t3} $\varrho=\frac{15}{32}r$, $d_*+\varrho<\tilde r$. For an orbit $\tilde y$ in $W^e_{\tilde r}$ with $\abs{\tilde I(0)-I_c}\le d_*$ violating the claim, Lemma~\ref{bpN:p7:exit-ball} gives $t^*$ with $\abs{\tilde I(t^*)-\tilde I(0)}=\varrho$ and $\tilde I(\tau)$ in the closed $\varrho$-ball before. There $\dot{\tilde I}=-\partial_\theta f_*(\tilde y)$ ($h,g$ are $\theta$-independent; the $I$-equation has no slow terms), $\abs{\dot{\tilde I}}\le\eta/(e\tilde s)=6\eta/(es)$ (Lemma~\ref{bpN:p11:real-system}(c)), so $\varrho\le\frac{6e^{-\kappa/6}\varepsilon}{es}\cdot\frac{sre^{\kappa/6}}{5\varepsilon}=\frac6{5e}r<0.442r<\varrho$: contradiction.
\proofstep{Transfer}\label{stepN:p11:nr-concl} Lemma~\ref{bpN:p11:transfer}: $\abs{I(u)-I_c}\le\frac{15}{32}r+\frac2{2^7}r=\frac{31}{64}r\le\frac r2$.
\end{proof}

\subsection{The resonant case: an $\omega$-free energy error on explicit domains}\label{bpN:p11:s-energy}

\begin{lemma}[Energy confinement near a resonance, convex case, no bound on $\omega$]\label{bpN:p11:energy-free}
Let $\tilde r,\tilde s>0$ (the analytic radius and width of the normal form), $I_c\in\R^n$, $\Lambda\subset\Z^n$, $\Pi=\Pi_\Lambda$ the Euclidean projection, $\Pi^\perp=1-\Pi$. Let $h$ be holomorphic on $V_{\tilde r}(I_c)$, real on real points, with on $U=\{\abs{I-I_c}<\tilde r\}$: $m\norm\xi_2^2\le Q(I)[\xi,\xi]$, $\norm{Q(I)}_{2\to2}\le M$ ($0<m\le M$). Let $g,f_*\in\Os{\tilde r}{\tilde s}^{e,\R}$ with finite norms, $P_\Lambda g=g$, $\zeta_g=\wn g{\tilde r}{\tilde s}$, $\eta=\wn{f_*}{\tilde r}{\tilde s}$, and let $\varepsilon,\bar\eta>0$ with
\[\zeta_g\le\tfrac54\varepsilon,\qquad\eta\le\bar\eta\le\varepsilon/e,\qquad\varepsilon\le\frac{m\tilde r^2}{2^{12}},\qquad16kc\tilde s\le\tilde r .\]
Let $\delta\in(0,m\tilde r/16]$ with $\norm{\Pi\omega(I_c)}_2\le\delta$, $\rho'=\delta/(4M)$ ($\le\tilde r/64$), $\tilde H=h+g+f_*$, $T_*=\dfrac{\tilde s\,m\,\tilde r}{20\,M\,\bar\eta}$. Then for every $\tilde H$-orbit $\tilde y$ in $W^e_{\tilde r}$ on $\tilde J\ni0$ with $\norm{\tilde I(0)-I_c}_2<\rho'$ \textbf{and} $\operatorname{dist}_\infty(\tilde x(t),\tilde S)<\tilde r/2$ for all $t\in\tilde J$ (\emph{explicit slow margin}):
\[\norm{\tilde I(t)-\tilde I(0)}_2\le\tilde r/4\qquad\text{for }t\in\tilde J,\ \abs t\le T_* .\]
\source{Pos93 pp.~197--198 (Lemma 1), Lemma~\ref{bpN:p7:no-exit}. \emph{New}: the identity \eqref{bpN:p11:eq:lie} replaces the factor $\norm{\omega_0}$; the Cauchy estimates are applied only at points of the strictly smaller tube $\Dom{\tilde r/2}{\tilde s/2}^e$, which the orbit does not leave on the stretch considered (review N11-01).}
\end{lemma}
\begin{proof}
Suppose some $t_1\in\tilde J$, $\abs{t_1}\le T_*$ violates the claim. Lemma~\ref{bpN:p7:exit-ball} gives $t^*$, $\abs{t^*}\le T_*$, with $\norm{\tilde I(\tau)-\tilde I(0)}_2\le\Delta:=\tilde r/4$ for $\tau$ between $0$ and $t^*$ and equality at $t^*$. Put $\omega_0=\omega(\tilde I(0))$, $\Delta I=\tilde I(t^*)-\tilde I(0)$.
\proofstep{The orbit stretch lies in the small tube}\label{stepN:p11:ef-tube} For $\tau$ between $0$ and $t^*$: $\abs{\tilde I(\tau)-I_c}\le\norm{\tilde I(\tau)-\tilde I(0)}_2+\norm{\tilde I(0)-I_c}_2<\frac{\tilde r}4+\frac{\tilde r}{64}<\frac{\tilde r}2$; $\operatorname{dist}(\tilde x(\tau),\tilde S)<\frac{\tilde r}2$ by hypothesis; $\theta$ is real. So $\tilde y(\tau)\in\Dom{\tilde r/2}{\tilde s/2}^e\cap\R^{2n+2k}$, and the segment $[\tilde I(0),\tilde I(t^*)]\subset U$. This is the only domain on which the Cauchy-type bounds below are used.
\proofstep{Admissibility}\label{stepN:p11:ef-adm} $\norm{\omega_0-\omega(I_c)}_2\le M\rho'=\delta/4$ (mean value inequality on the convex ball $\{\norm{I-I_c}_2<\rho'\}\subset U$), so $\norm{\Pi\omega_0}_2\le\frac54\delta$; $\rho'\le\frac{m\tilde r}{64M}\le\frac{\tilde r}{64}$.
\proofstep{Energy inequality}\label{stepN:p11:ef-en} The inequality is proved directly in the extended phase space, in three substeps. Lemma~\ref{bpN:p7:energy-core} is \emph{not} applied: it is stated for a canonical orbit in $\R^n\times\R^n$ of a Hamiltonian $h_\R(I)+g+f$ with $g,f$ functions of $(\theta,I)$ only, whereas $\tilde y$ is an orbit of $X^c_{\tilde H^\R}$ in $\R^{2n+2k}$ and $g,f_*$ also depend on the slow pair $\tilde x$. Only its argument is repeated, and only along the fast action segment. Write $h_\R=h|_U$ (holomorphic on $V_{\tilde r}(I_c)$, real on real points, hence $C^\infty$ on $U$ by \ref{bpN:p1:ext-restrict}, exactly as in Step~\ref{stepN:p12:fb-real}(i), (ii)), $\omega=\nabla h_\R$, $Q=D^2h_\R$, $\omega_0=\omega(\tilde I(0))$, $I_\sigma=\tilde I(0)+\sigma\Delta I$ ($0\le\sigma\le1$), $\Delta=\norm{\Delta I}_2=\tilde r/4$.

\proofstep{Energy}\label{stepN:p11:ef-en-energy} $\tilde y$ is an $\tilde H$-orbit in $W^e_{\tilde r}$ on an interval containing $0$ and $t^*$, so $\tilde H^\R\circ\tilde y$ is constant there (Lemma~\ref{bpN:p11:real-system}(b)). Since $\tilde H^\R=h_\R(I)+(g+f_*)^\R$ on $W^e_{\tilde r}$ and $h_\R$ does not depend on $\theta,\tilde x$,
\[h_\R(\tilde I(t^*))-h_\R(\tilde I(0))=-\bigl[(g+f_*)(\tilde y(t^*))-(g+f_*)(\tilde y(0))\bigr],\]
whose modulus is $A\le2(\zeta_g+\eta)$ because $\abs{u(y)}\le\wn u{\tilde r}{\tilde s}$ at real points $y\in W^e_{\tilde r}$ (Lemma~\ref{bpN:p11:real-system}(c)).

\proofstep{Taylor along the fast segment}\label{stepN:p11:ef-en-taylor} $I_\sigma\in U$ for $\sigma\in[0,1]$ (Step~\ref{stepN:p11:ef-tube}), so $\varphi(\sigma)=h_\R(I_\sigma)$ is $C^2$ on $[0,1]$ with $\varphi'(0)=\ip{\omega_0}{\Delta I}$ and $\varphi''(\sigma)=Q(I_\sigma)[\Delta I,\Delta I]\ge m\Delta^2$ (hypothesis on $U$). The Lagrange form of Taylor's formula (\ref{bpN:p7:ext-taylor}) gives $\varphi(1)-\varphi(0)-\varphi'(0)=\frac12\varphi''(\xi)\ge\frac m2\Delta^2$. By the energy identity, $\varphi(1)-\varphi(0)\le A$; hence $\frac m2\Delta^2\le A-\ip{\omega_0}{\Delta I}$.

\proofstep{Splitting}\label{stepN:p11:ef-en-split} $\Pi$ is self-adjoint with $\Pi+\Pi^\perp=1$ (\ref{bpN:p7:ext-linalg}), so $\ip{\omega_0}{\Delta I}=\ip{\Pi\omega_0}{\Delta I}+\ip{\omega_0}{\Pi^\perp\Delta I}$, and therefore
$\frac m2\Delta^2\le A+\abs{\ip{\Pi\omega_0}{\Delta I}}+C$, $C=\abs{\ip{\omega_0}{\Pi^\perp\Delta I}}$.

 Now $\abs{\ip{\Pi\omega_0}{\Delta I}}\le\frac54\delta\Delta\le\frac5{64}m\tilde r\Delta=\frac5{16}m\Delta^2$ ($\tilde r=4\Delta$). Hence $\frac3{16}m\Delta^2\le A+C$, and $\frac3{16}m\Delta^2=\frac{3m\tilde r^2}{256}>0.0117\,m\tilde r^2$.
\proofstep{The identity}\label{stepN:p11:ef-id} Along $\tilde y$: $\mathsf F(\tau)=f_*(\tilde y(\tau))$ has $\mathsf F'=Df_*\cdot X^c_{\tilde H^\R}=\{f_*,\tilde H\}_c=\{f_*,h\}_c+\{f_*,g\}_c$ ($\{f_*,f_*\}_c=0$), and $\{f_*,h\}_c=\partial_\theta f_*\cdot\omega(\tilde I)$ since $h$ depends only on $I$:
\begin{equation}\label{bpN:p11:eq:lie}\partial_\theta f_*\cdot\omega(\tilde I(\tau))=\mathsf F'(\tau)-\{f_*,g\}_c(\tilde y(\tau)).\end{equation}
\proofstep{Estimate of $C$}\label{stepN:p11:ef-C} $\dot{\tilde I}=-\partial_\theta f_*-\partial_\theta g$, $\Pi^\perp\partial_\theta g=0$ (Lemma~\ref{bpN:p11:real-system}(d)), so $\Pi^\perp\Delta I=-\int_0^{t^*}\Pi^\perp\partial_\theta f_*\,d\tau$ and, $\Pi$ being self-adjoint, $\ip{\omega_0}{\Pi^\perp\Delta I}=-\int\ip{\omega_0}{\partial_\theta f_*}+\int\ip{\Pi\omega_0}{\partial_\theta f_*}$; with $\ip{\omega_0}{\partial_\theta f_*}=\ip{\omega(\tilde I)}{\partial_\theta f_*}+\ip{\omega_0-\omega(\tilde I)}{\partial_\theta f_*}$ and \eqref{bpN:p11:eq:lie}:
\[C\le\abs{\mathsf F(t^*)-\mathsf F(0)}+\abs{t^*}\Bigl(\sup\abs{\{f_*,g\}_c}+M\Delta\sup\norm{\partial_\theta f_*}_2+\tfrac54\delta\sup\norm{\partial_\theta f_*}_2\Bigr),\]
suprema over $\tau$ between $0$ and $t^*$. $\abs{\mathsf F(t^*)-\mathsf F(0)}\le2\eta$. \emph{Angular derivative:} at real points $\norm{\partial_\theta f_*}_2\le\eta/(e\tilde s)$ (Lemma~\ref{bpN:p11:real-system}(c)). \emph{Bracket:} Lemma~\ref{bpN:p11:bracket} with $(u,v)=(f_*,g)$, $R_0=R=\tilde r$, losses $(\rho,\sigma)=(\tilde r/2,\tilde s/2)$, whose $\textup{(SL)}_{\rho,\sigma}$ is $16kc\tilde s\le\tilde r$, gives $\wn{\{f_*,g\}_c}{\tilde r/2}{\tilde s/2}\le\frac1{\rho\sigma}\eta\zeta_g=\frac{4\eta\zeta_g}{\tilde r\tilde s}$; by Step~\ref{stepN:p11:ef-tube} the points $\tilde y(\tau)$ lie in $\Dom{\tilde r/2}{\tilde s/2}^e$, so $\sup\abs{\{f_*,g\}_c}\le\frac{4\eta\zeta_g}{\tilde r\tilde s}\le\frac{5\eta\varepsilon}{\tilde r\tilde s}\le\frac{5m\tilde r}{2^{12}\tilde s}\eta$. With $M\Delta=M\tilde r/4$, $\frac54\delta\le\frac5{64}m\tilde r$, $\frac1e<0.3679$ and $m\le M$, and the exact coefficient
\[\frac5{2^{12}}+\frac5{64}\cdot\frac1e+\frac14\cdot\frac1e<0.00123+0.02875+0.09198<0.1220 ,\]
\[C\le2\eta+0.1220\,\frac{M\tilde r\,\eta\,\abs{t^*}}{\tilde s}\le2\bar\eta+0.0061\,m\tilde r^2\qquad(\abs{t^*}\le T_*=\tfrac{\tilde sm\tilde r}{20M\bar\eta},\ \eta\le\bar\eta).\]
\proofstep{Contradiction}\label{stepN:p11:ef-contra} $A\le2(\frac54+\frac1e)\varepsilon<3.24\varepsilon\le0.00080\,m\tilde r^2$, $2\bar\eta\le\frac{2\varepsilon}e<0.00018\,m\tilde r^2$ ($\varepsilon\le m\tilde r^2/2^{12}$); so $A+C<(0.00080+0.00018+0.0061)m\tilde r^2<0.0071\,m\tilde r^2<0.0117\,m\tilde r^2$. Contradiction.
\end{proof}

\subsection{The stopped block theorem}\label{bpN:p11:s-block}

\begin{lemma}[Hessian and sizes (nonempty $G$)]\label{bpN:p11:hessian}
Let $h$ be as in Theorem~\ref{bpN:p0:partial}, $\sigma$ as in \texttt{HasDefiniteFastPart}, and assume $G\ne\emptyset$. For $z\in G'_\rho$ the symmetric matrix $Q_{\C}(z)=(D^2h(z)[e_i,e_j])_{ij}$ has $\norm{Q_{\C}(z)}_{2\to2}\le M_{\C}$. For real $I\in U$ (the real slice of $G'_\rho$): $Q(I)=D^2h_\R(I)=Q_{\C}(I)$ (Lemma~\ref{bpN:p1:holo-real}), $m\norm\xi_2^2\le\sigma Q(I)[\xi,\xi]$, $\abs{Q(I)[\xi,\xi]}\le M_{\C}\norm\xi_2^2$, $\abs{\omega(I)}_\infty\ge g$. Consequently $m\le M_{\C}$ (take $I^*\in G\subset G'\subset U$ and any unit $\xi$). For $h_1=h/2\pi$ the same with $(m_1,M_1,g_1)=(m,M_{\C},g)/2\pi$, and $\sigma h_1$ is $m_1$-convex.
\end{lemma}
\begin{proof}
\proofstep{Operator norm}\label{stepN:p11:hs-op} $\abs{D^2h(z)[u,v]}\le M_{\C}\abs u_\infty\abs v_\infty\le M_{\C}\norm u_2\norm v_2$ and $\norm{Q_{\C}u}_2=\sup_{\norm v_2=1}\abs{D^2h(z)[u,\bar v]}$. $G\subset G'$: $I\in\bar B(I,\rho)\subset G'$ and a real $I\in G'$ lies in $G'_\rho$. The rest is the definition of \texttt{HasDefiniteFastPart}, and $\abs\xi_\infty\le\norm\xi_2$. The inequality $m\le\sigma Q[\xi,\xi]\le M_{\C}$ at a real point of $G$ for a unit vector gives $m\le M_{\C}$; this needs a point of $G$.
\end{proof}

\begin{definition}[Sets and the extended system for the block theorem]\label{bpN:p11:block-setting}
Data and constants of Definition~\ref{bpN:p11:constants}, $G\ne\emptyset$, $\mu>0$ with $\varepsilon=\varepsilon(\mu)\le\varepsilon_0$ and $d=\mu^{b_n}\le\rho$. $S=\operatorname{core}(D,3d/4)\subset\R^{2k}$, $w=d/4$. The \emph{perturbation pair} is $(\breve h,\breve f)=(\sigma h_1,\sigma f_1)$, $f_1(\theta,I,x)=\frac\mu{2\pi}P\bigl((2\pi\theta,q),(I,p)\bigr)$, $x=(q,p)$. For $I_c\in G$ and a block starting radius $r_B\le r_0$: $\lambda=w/r_B$, $c=2\pi/\lambda^2$, $W^e_\varrho$ as in \S\ref{bpN:p11:s-real}, and the physical real set $\mathcal W(I_c,r_B)=\{(\theta,I;x):\abs{I-I_c}<r_B,\ \operatorname{dist}_\infty(x,S)<w\}$.
\end{definition}

\begin{lemma}[Analyticity, norm and field on the block domain]\label{bpN:p11:domain}
In Definition~\ref{bpN:p11:block-setting}, $I_c\in G$, $0<r_B\le r_0$. (a) $\breve h$ is holomorphic on $V_{r_B}(I_c)\subset G'_\rho$, real on real points, $\norm{\partial_I^2\breve h}_{2\to2}\le M_1$, $m_1$-convex on $U_{r_B}(I_c)$, $\abs{\omega}_\infty\ge g_1$. (b) $\breve f\in\Os{r_B}{s}^{e,\R}$ and $\wn{\breve f}{r_B}s\le\varepsilon(\mu)=\kappa_C\mu/(2\pi)$. (c) On the open set $\mathcal W(I_c,r_B)$ the frozen $\hat H_\mu=\frac1{2\pi}H_\mu\circ\Theta^{-1}$ equals $\operatorname{Re}(h_1+f_1)$ and is real analytic, so $\sigma X^{2\pi}_{\hat H_\mu}=X^{2\pi}_{(\breve h+\breve f)^\R}$ there (sign $\sigma$); in $\tilde x=x/\lambda$ this is $X^c_{(\breve h+\breve f)^\R}$ on $W^e_{r_B}$.
\end{lemma}
\begin{proof}
\proofstep{Complex points}\label{stepN:p11:dm-pts} Let $(\theta,I,x)$ be complex with $\abs{\operatorname{Im}\theta}<2s=\frac\rho{2\pi}$, $\abs{I-I_c}<r_B\le\frac\rho2$, $\operatorname{dist}_\infty(x,S)<w$. Then $z=((2\pi\theta,q),(I,p))\in\mathcal F(G',D,\rho)$: $\abs{\operatorname{Im}2\pi\theta_i}<\rho$; $\operatorname{Re}I$ is within $\rho/2$ of $I_c\in G$ so lies in $G'$ (collar); $\abs{\operatorname{Im}I_i}<\rho/2$; with $x_0\in S$, $\abs{x-x_0}<w$, $\operatorname{Re}x\in\bar B(x_0,d/4)\subset D$ ($\bar B(x_0,3d/4)\subset D$); $\abs{\operatorname{Im}x_j}<w<\rho$. So $f_1$ is holomorphic (P1), $\abs{f_1}\le\frac\mu{2\pi}$ (P2), $1$-periodic in $\theta$ (P4), real at real points (P3). The same points give $V_{r_B}(I_c)\subset G'_\rho$; (a) follows from Lemma~\ref{bpN:p11:hessian}.
\proofstep{Norm}\label{stepN:p11:dm-norm} B.1(b) (Lemma~\ref{bpN:p11:ext-part4}) with $\sigma=s$: $\wn{f_1}{r_B}s\le\coth(\pi s)^n\frac\mu{2\pi}$ (the bound holds on $\Dom{r_B}{2s}^e$).
\proofstep{Field}\label{stepN:p11:dm-field} $\mathcal W(I_c,r_B)\subset\{\mathrm{cx}\,z\in\mathcal F\}$ (open); on its real points $H_\mu=\operatorname{Re}(h+\mu P\circ\mathrm{cx})$, so $\hat H_\mu$ is the real part of a holomorphic function and Lemma~\ref{bpN:p11:real-system}(a) applies; rescaling $x=\lambda\tilde x$ turns weight $2\pi$ into $c$ (Definition~\ref{bpN:p11:ext-class}).
\end{proof}

\begin{lemma}[Block data from the covering lemma]\label{bpN:p11:block-data}
In Definition~\ref{bpN:p11:block-setting} assume $\kappa=2\pi K_as\ge6$ ($K_a=K/\sqrt n$, $K=(\varepsilon_0/\varepsilon)^{1/(2n)}\ge1$). Let $I_0\in G$, $p=\frac98$, $q=9$, $b_{\rm c}=\frac1{32}$. (1) $r_\Theta:=r_0/(AK)^n\le r\le R_0/K$; (2) $\abs{k\cdot\omega(I_0)}\ge\alpha$ for $k\in Z_{K_a}\setminus\Lambda$ (for $\Lambda=\Theta$: $0<\abs k_1\le K_a$, with $\alpha_\Theta$); (3) $\norm{\Pi_\Lambda\omega(I_0)}_2<\frac{m_1r}{32}$ (for $\Lambda=\Theta$ the projection is $0$, so this is the trivial $0<m_1r/32$, not the nontrivial-lattice closeness clause); (4) \ref{bpN:p6:cond-L} holds on $V_r(I_0)$ with $L=\sqrt nM_1$ and $K_aLr=\alpha/p$; (5) $\varepsilon\le m_1r^2/2^{14}$ and $\varepsilon\le\alpha r/(2^7qK_a)$.
\end{lemma}
\begin{proof}
\proofstep{Covering}\label{stepN:p11:bd-cover} Lemma~\ref{bpN:p8:covering}(a),(d) (valid for $K\ge1$): $\Lambda\in\mathbb M_K$, $I_0\in D_\Lambda$, $r_\Theta\le r_\Lambda\le r_0/(AK)=R_0/K$ for $\Lambda\ne\Z^n$, $r_{\Z^n}=r_0$. If $\Lambda=\Z^n$ then $\Pi_\Lambda=1$ and (c) gives $\norm{\omega(I_0)}_2<\delta_{\Z^n}=b_{\rm c}m_1r_0=\frac{m_1r_0}{32}\le g_1$ (as $r_0\le32g/m$), contradicting $\norm{\omega(I_0)}_2\ge\abs{\omega(I_0)}_\infty\ge g_1$ (Lemma~\ref{bpN:p11:hessian}): so $\Lambda\ne\Z^n$.
\proofstep{(2)--(4)}\label{stepN:p11:bd-24} Corollary~\ref{bpN:p8:cutoffs}(b),(c) with $K_g=K$, $K_a=K/\sqrt n\le K_g$: (2), (3) ($\delta_\Lambda=b_{\rm c}m_1r$). Lemma~\ref{bpN:p6:lipschitz}-type: for $v\in\C^n$, $\abs{\partial_I^2h_1v}_\infty\le\norm{\cdot}_2\le M_1\norm v_2\le\sqrt nM_1\abs v_\infty$. $\alpha=pM_1Kr=p\sqrt nM_1K_ar$, i.e.\ $K_aLr=\alpha/p$. \emph{Sign transport:} the lemma is applied to $\omega_{\sigma h_1}=\sigma\omega_{h_1}$; all quantities used (absolute divisors $\abs{k\cdot\omega}$, lengths $\norm{\Pi\omega}_2$, $\norm\omega_2\ge g_1$, Hessian operator norms) are unchanged by $\omega\mapsto\sigma\omega$, and $\omega_{\sigma h_1}$ is real on the real slice, as required by Proposition~\ref{bpN:p11:hom}.
\proofstep{(5)}\label{stepN:p11:bd-5} $m_1r_\Theta^2/2^{14}=\varepsilon_0K^{-2n}=\varepsilon$ ($K^{2n}=\varepsilon_0/\varepsilon$) and $r\ge r_\Theta$; $\frac{\alpha r}{2^7qK_a}=\frac{p\sqrt nM_1r^2}{2^7q}\ge\frac{M_1r^2}{2^{10}}\ge\frac{m_1r^2}{2^{14}}\ge\varepsilon$.
\end{proof}

\begin{theorem}[Stopped block stability (Pos93 Theorem 1 with slow parameters)]\label{bpN:p11:block}
In Definition~\ref{bpN:p11:block-setting} assume (B1) $\kappa=2\pi K_as\ge6$ and (B2) $K\ge256\,k\,\rho\,R_0/d^2$. Put $r_\Theta=r_0/(AK)^n$, $T_{\mathrm P}=\dfrac{s\,m_1\,r_\Theta}{240\,M_1\,\varepsilon}e^{\kappa/6}$. Let $I_0\in G$, $F:\R^{2n+2k}\to\R^{2n+2k}$ any function with $F=X^{2\pi}_{(\breve h+\breve f)^\R}$ on $\mathcal W(I_0,r_0)$, $J\ni0$ an interval, $y=(\theta,I;x)$ differentiable on $J$ with $y'=F(y)$ there, $I(0)=I_0$, $x(0)\in S$, and $t_1\in J$ with $\abs{t_1}\le T_{\mathrm P}$ and $x(u)\in\operatorname{core}(D,d)$ for all $u$ between $0$ and $t_1$. Then $\abs{I(u)-I_0}_\infty\le\frac{R_0}{2K}$ for all $u$ between $0$ and $t_1$.
\source{Pos93 Theorem 1 (pp.~190--191, proof pp.~203--204); the covering is Part~\ref{bpN:p8} with $b_{\rm c}=\frac1{32}$ in place of Pos93's $\frac18$ (Remark~\ref{bpN:p11:b-change}).}
\end{theorem}
\begin{proof}
We write $h,f$ for $\breve h,\breve f$.
\proofstep{Block}\label{stepN:p11:bl-block} Lemma~\ref{bpN:p11:block-data}: $\Lambda$, $r=r_\Lambda\in[r_\Theta,R_0/K]$, $\alpha$. Take $r_B=r$, $\lambda=w/r$, $c=2\pi r^2/w^2$.
\proofstep{(SL)}\label{stepN:p11:bl-sl} $64\pi ksr\le64\pi ks\frac{R_0}K=16k\rho\frac{R_0}K\le\frac{d^2}{16}=w^2$ by (B2) ($s=\rho/4\pi$). This is the only place where $d$, hence $b_n$, enters.
\proofstep{Extended system}\label{stepN:p11:bl-ext} By Lemma~\ref{bpN:p11:domain}: $h$ on $V_r(I_0)$, $f\in\Os rs^{e,\R}$ with $\wn frs\le\varepsilon$; $\tilde y=(\theta,I,x/\lambda)$ satisfies $\tilde y'=F^\sim(\tilde y)$ with $F^\sim=X^c_{(h+f)^\R}$ on $W^e_r\subset$ image of $\mathcal W(I_0,r_0)$; $\tilde x(u)\in\tilde S$ since $\operatorname{core}(D,d)\subset S$.
\proofstep{Case $\Lambda=\Theta$}\label{stepN:p11:bl-theta} Proposition~\ref{bpN:p11:nonresonant} with $(r,K_a,\alpha,L)\to(r_\Theta\text{-block: }r,K_a,\alpha_\Theta,\sqrt nM_1)$, $p=\frac98$: hypotheses by Lemma~\ref{bpN:p11:block-data}(2),(4),(5), (B1), (SL); $T_{\rm nr}=\frac{sr}{5\varepsilon}e^{\kappa/6}\ge T_{\mathrm P}$ ($r=r_\Theta$ for $\Theta$; $\frac{m_1}{M_1}\le1$). Conclusion $\abs{I(u)-I_0}\le r/2\le R_0/(2K)$.
\proofstep{Case $\Lambda\ne\Theta,\Z^n$}\label{stepN:p11:bl-res} Proposition~\ref{bpN:p11:resonant} with $(r,K_a,\alpha,m,M)=(r_\Lambda,K_a,\alpha_\Lambda,m_1,M_1)$: hypotheses by Lemma~\ref{bpN:p11:block-data}(2)--(5), (B1), (SL), $m_1$-convexity (Lemma~\ref{bpN:p11:domain}(a)); $T_\Lambda=\frac{sm_1r_\Lambda}{240M_1\varepsilon}e^{\kappa/6}\ge T_{\mathrm P}$ since $r_\Lambda\ge r_\Theta$. \end{proof}

\subsection{Assembly: proof of the frozen theorem}\label{bpN:p11:s-final}

\begin{proof}[Proof of Theorem~\ref{bpN:p0:partial}]
\proofstep{Empty $G$}\label{stepN:p11:final-empty} If $G=\emptyset$ then $\operatorname{core}(G,\cdot)=\emptyset$ and (Z2) is impossible: any $\mu_0>0$ (say $1$) works. From now on $G\ne\emptyset$; Lemma~\ref{bpN:p11:hessian} (in particular $m\le M_{\C}$) holds.
\proofstep{Data and constants}\label{stepN:p11:final-data} Definition~\ref{bpN:p11:constants} ($A=40M_{\C}/m\ge11$, $\varepsilon_0=m_1r_0^2/(2^{14}A^{2n})$, $r_0=\min\{\rho/2,32g/m\}$, $R_0=r_0/A$, $c_K=(2\pi\varepsilon_0/\kappa_C)^{1/(2n)}$).
\proofstep{The threshold}\label{stepN:p11:final-mu} With $\gamma_1,\gamma_2$ of Lemma~\ref{bpN:p11:exponents} let $\mu_0=\min\{\mu_a,\dots,\mu_f\}$: $\mu_a=2\pi\varepsilon_0/\kappa_C$; $\mu_b=\rho^{1/b_n}$; $\mu_c=(\pi sc_K/(3\sqrt n))^{2n}$; $\mu_d=(c_K/(256k\rho R_0))^{1/\gamma_1}$; $\mu_e=(c_K/R_0)^{1/\gamma_2}$; $\mu_f$ from Step~\ref{stepN:p11:final-time}. For $0<\mu<\mu_0$: $\varepsilon\le\varepsilon_0$; $d=\mu^{b_n}\le\rho$; $\kappa=2\pi c_Ks\mu^{-1/(2n)}/\sqrt n\ge6$ (B1); $K=c_K\mu^{-1/(2n)}\ge256k\rho R_0\mu^{-2b_n}$ iff $\mu^{\gamma_1}\le c_K/(256k\rho R_0)$ (B2); $\frac{R_0}{2K}\le\frac12\mu^{b_n}$ iff $\mu^{\gamma_2}\le c_K/R_0$.
\proofstep{Time}\label{stepN:p11:final-time} $T_{\mathrm P}=C_T\mu^{-1/2}\exp(c_2\mu^{-1/(2n)})$ with $C_T=\frac{s\,m\,r_0}{240\,M_{\C}(Ac_K)^n}\cdot\frac{2\pi}{\kappa_C}>0$ and $c_2=\frac{\pi sc_K}{3\sqrt n}>0$ ($r_\Theta=r_0(Ac_K)^{-n}\mu^{1/2}$, $\varepsilon=\kappa_C\mu/2\pi$, $m_1/M_1=m/M_{\C}$). As $a_n<\frac1{2n}$, $c_2\mu^{-1/(2n)}-\mu^{-a_n}\ge\frac{c_2}2\mu^{-1/(2n)}$ for small $\mu$, so $\mu T_{\mathrm P}e^{-\mu^{-a_n}}\ge C_T\mu^{1/2}e^{\frac{c_2}2\mu^{-1/(2n)}}\to\infty$ (\ref{bpN:p11:ext-rpow}); this defines $\mu_f$.
\proofstep{Given $\mu,P,C,z,t$}\label{stepN:p11:final-given} If $t=0$ the claim is $0\le\frac12\mu^{b_n}$. Let $t>0$; $t\le\mu^{-1}e^{\mu^{-a_n}}\le T_{\mathrm P}$, $[0,t]\subset[0,C]$.
\proofstep{The orbit}\label{stepN:p11:final-orbit} By (Z1), $z$ is differentiable on $[0,C]$ with $z'=X_{H_\mu}(z)$; $I_0:=\mathrm{fast}(z(0)_2)\in G$ by (Z2). $y=\Theta\circ z$ satisfies $y'=F_0(y)$ with $F_0=X^{2\pi}_{\hat H_\mu}$, an everywhere defined function (Lemma~\ref{bpN:p11:orbit}); by Lemma~\ref{bpN:p11:domain}(c), $\sigma F_0=X^{2\pi}_{(\breve h+\breve f)^\R}$ on $\mathcal W(I_0,r_0)$.
