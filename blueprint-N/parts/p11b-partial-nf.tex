\subsection{The extended class: slow variables as parameters}\label{bpN:p11:s-ext-class}

Throughout \S\ref{bpN:p11:s-ext-class}--\ref{bpN:p11:s-nf}: $n\ge1$, $k\ge1$, $I_c\in\R^n$ is a fixed base action, $S\subset\R^{2k}$ a fixed set of physical slow pairs $x=(q,p)$ (ordered $q_1..q_k,p_1..p_k$; later $S=\operatorname{core}(D,3d/4)$). A \emph{block} fixes once and for all a \emph{starting radius} $r_B>0$, a physical slow width $w>0$ and
\[\lambda:=\frac w{r_B},\qquad c:=\frac{2\pi}{\lambda^2}=\frac{2\pi r_B^2}{w^2},\qquad \tilde x:=x/\lambda,\qquad\tilde S:=\lambda^{-1}S .\]
$\lambda$ and $c$ are \emph{never recomputed} during an iteration: an iterated (smaller) radius $R\le r_B$ has physical slow tube width $\lambda R$.

\begin{definition}[Extended class $\Os Rs^{e}$, extended phase space, bracket]\label{bpN:p11:ext-class}
$\hat D=\{I_c\}\times\tilde S\subset\C^n\times\C^{2k}$; extended actions $\hat I=(I,\tilde x)\in\C^{n+2k}$ (so $m=n+2k$ in Part~\ref{bpN:p4}); points $y=(\theta,I,\tilde x)\in\C^{2n+2k}$. For $0<R\le r_B$:
$\Dom Rs^{e}=T_s\times V_R\hat D=\{\abs{\operatorname{Im}\theta}<s,\ \abs{I-I_c}<R,\ \operatorname{dist}_\infty(\tilde x,\tilde S)<R\}$ (physically $\operatorname{dist}_\infty(x,S)<\lambda R$), and $\Os Rs^e$, $\wn uRs$ are the class and norm of Definition~\ref{bpN:p4:norm} for $D=\hat D$; $\Os Rs^{e,\R}$ those with real values on $W^e_R:=\Dom Rs^e\cap\R^{2n+2k}$. The bracket is $\{u,v\}_c$ of Definition~\ref{bpN:p11:wfield} in the variables $(\theta,I,\tilde x)$, $\tilde x=(\tilde q,\tilde p)$; since $x=\lambda\tilde x$ and $c\lambda^2=2\pi$ it is the weight-$2\pi$ bracket of the physical slow pair. The conditions
\[\textup{(SL)}_{\rho,\sigma}:\ \ 16kc\sigma\le\rho,\qquad\qquad\textup{(SL)}:\ \ c\le\frac{r_B}{32ks}\ \Bigl(\iff64\pi ksr_B\le w^2\Bigr)\]
are used; (SL) implies $\textup{(SL)}_{\rho,\sigma}$ whenever $\sigma\le2s\rho/r_B$.
\source{Ours (device for reusing Part~\ref{bpN:p4}); Nek77 Lemma 10.3 (p.~45--46) has the same tube with a slow loss $b$.}
\end{definition}

\begin{lemma}[Part~\ref{bpN:p4} for the extended class]\label{bpN:p11:ext-part4}
Lemmas~\ref{bpN:p4:coefficients}, \ref{bpN:p4:sup-weighted}, \ref{bpN:p4:algebra}, \ref{bpN:p4:cauchy} ((a) pointwise in $\hat I$; (b) with $\max_{i\le n+2k}$), \ref{bpN:p4:general-cauchy} and \ref{bpN:p4:projections} hold for $\Os\cdot s^e$ with the same constants. \emph{Proof:} Part~\ref{bpN:p4} is proved for an arbitrary $D\subset\C^m$ with $m$ independent of $n$ (its introduction), and $\hat D\subset\C^{n+2k}$ is such a set; only the bracket uses $m=n$. \qed
\end{lemma}

\begin{lemma}[Bracket estimate with slow pairs, B.4$^e$]\label{bpN:p11:bracket}
Let $u\in\Os{R_0}{s_0}^e$, $v\in\Os Rs^e$ have finite norms, $0<\rho<R$, $0<\sigma<s$, $R-\rho<R_0$, $s-\sigma<s_0$. Then $\{u,v\}_c\in\Os{R-\rho}{s-\sigma}^e$ and
\[\wn{\{u,v\}_c}{R-\rho}{s-\sigma}\le\Bigl[\frac1e\Bigl(\frac1{(R_0-R+\rho)\sigma}+\frac1{(s_0-s+\sigma)\rho}\Bigr)+\frac{2kc}{(R_0-R+\rho)\rho}\Bigr]\wn u{R_0}{s_0}\wn vRs .\]
Under $\textup{(SL)}_{\rho,\sigma}$ the bracket in $[\ ]$ is $\le\frac12\bigl(\frac1{(R_0-R+\rho)\sigma}+\frac1{(s_0-s+\sigma)\rho}\bigr)$. The pointwise values on $\Dom{R-\rho}{s-\sigma}^e$ are bounded by the weighted norm (Lemma~\ref{bpN:p4:sup-weighted}(a)).
\end{lemma}
\begin{proof}
\proofstep{Fast part}\label{stepN:p11:br-fast} Lemma~\ref{bpN:p4:bracket} (its proof uses only Lemmas~\ref{bpN:p4:algebra}, \ref{bpN:p4:cauchy}).
\proofstep{Slow part}\label{stepN:p11:br-slow} Fix $\hat I\in V_{R-\rho}\hat D$, $t=s-\sigma$. Each of the $2k$ products $\partial_{\tilde q_j}u\,\partial_{\tilde p_j}v$, $\partial_{\tilde p_j}u\,\partial_{\tilde q_j}v$ has $\wn{\cdot}{\hat I}{t}\le\wn{\partial u}{\hat I}t\wn{\partial v}{\hat I}t\le\frac{\wn u{R_0}{s_0}}{R_0-R+\rho}\cdot\frac{\wn vRs}\rho$ (Lemma~\ref{bpN:p4:algebra} pointwise, Lemma~\ref{bpN:p4:cauchy}(b)); multiply by $c$, sum, take $\sup_{\hat I}$ last.
\proofstep{Domination}\label{stepN:p11:br-dom} $\frac{2kc}{(R_0-R+\rho)\rho}=\frac{2kc\sigma}\rho\cdot\frac1{(R_0-R+\rho)\sigma}\le\frac18\cdot\frac1{(R_0-R+\rho)\sigma}$ and $\frac1e+\frac18<0.493<\frac12$.
\end{proof}

\begin{lemma}[Bracket with a decomposed $g$]\label{bpN:p11:bracket-g}
In the situation of Lemma~\ref{bpN:p6:bracket-g} for $\Os\cdot\cdot^e$ and $\textup{(SL)}_{\rho,\sigma}$: $\wn{\{g,\phi\}_c}{R-\rho}{s-\sigma}\le\frac1{\rho\sigma}\abs g_*\wn\phi Rs$ ($\abs g_*$ as in Definition~\ref{bpN:p6:setting}, for the extended norms).
\end{lemma}
\begin{proof}
Lemma~\ref{bpN:p11:bracket} for each $g_j$: $\le\frac12\bigl(\frac1{(R_j-R+\rho)\sigma}+\frac1{(s_j-s+\sigma)\rho}\bigr)\wn{g_j}{}{}\wn\phi{}{}=\frac1{\rho\sigma}\cdot\frac12\bigl(\frac\rho{R_j-R+\rho}+\frac\sigma{s_j-s+\sigma}\bigr)\wn{g_j}{}{}\wn\phi{}{}$; sum over $j$.
\end{proof}

\subsection{The Lie transform for the extended class}\label{bpN:p11:s-lie}

\begin{definition}[Scaled norm, field]\label{bpN:p11:scaled}
$\nu_{\rho,\sigma}(a,b,\xi)=\max(\abs a/\sigma,\abs b/\rho,\abs\xi/\rho)$ on $\C^n\times\C^n\times\C^{2k}$. $X^c_\phi=\Omega_c\nabla\phi$ has components $\partial_I\phi$, $-\partial_\theta\phi$, $c\,\partial_{\tilde p}\phi$, $-c\,\partial_{\tilde q}\phi$. The closed $\nu$-ball of radius $a$ about $y\in\Dom Rs^e$ is the polydisc $\bar P(y;a\sigma,a\rho,a\rho)$; for $a\le1$, $\sigma<s$, $\rho<R$... (Definition~\ref{bpN:p4:domains}(b) for $\hat D$): if $y\in\Dom{R-\rho}{s-\sigma}^e$ and $a\le1$ then $\bar B_\nu(y,a)\subset\Dom{R-(1-a)\rho}{s-(1-a)\sigma}^e$.
\end{definition}

\begin{lemma}[Size of the field]\label{bpN:p11:vf}
Let $\phi\in\Os Rs^e$ have finite norm, $0<\rho<R$, $0<\sigma<s$, $\textup{(SL)}_{\rho,\sigma}$. On $\Dom{R-\rho}{s-\sigma}^e$: $\abs{\partial_{I_i}\phi}\le\wn\phi Rs/\rho$, $\abs{\partial_{\theta_i}\phi}\le\wn\phi Rs/(e\sigma)$, $\abs{\partial_{\tilde x_j}\phi}\le\wn\phi Rs/\rho$, and $\nu_{\rho,\sigma}(X^c_\phi)\le\wn\phi Rs/(\rho\sigma)$; moreover $c\wn\phi Rs/\rho\le\wn\phi Rs/(16k\sigma)$.
\end{lemma}
\begin{proof}
Lemma~\ref{bpN:p11:ext-part4} (B.3) and B.1(a) as in Lemma~\ref{bpN:p5:vf-bound}; $c\le\rho/(16k\sigma)$ gives the slow bounds: $c\abs\phi/\rho\le\abs\phi/(16k\sigma)$, $\nu$-slow $\le c\abs\phi/\rho^2\le\abs\phi/(16k\rho\sigma)$.
\end{proof}

\begin{lemma}[Flow package]\label{bpN:p11:flow}
Lemma~\ref{bpN:p5:flow} applies verbatim to $N=2n+2k$, the norm $\nu_{\rho,\sigma}$, the open set $\Omega=\Dom{R'}{s'}^e$ and a holomorphic field $F$ with $\nu(F)\le\varepsilon\le\frac12$ on $\Omega$; (d) (translations) with $\tau=(\ell,0,0)$, $\ell\in\Z^n$; (e) (reality) with the field real on $\Omega\cap\R^{2n+2k}$ (the proof is the conjugation symmetry and does not use $N=2n$).
\end{lemma}
\begin{proof}
Lemma~\ref{bpN:p5:flow} is stated for an arbitrary complex norm on $\C^N$ and an arbitrary open $\Omega$; the reality statement is proved by complex conjugation of the solution and uniqueness. \qed
\end{proof}

\begin{lemma}[Symplectic flow; canonical maps]\label{bpN:p11:symplectic}
In Lemma~\ref{bpN:p11:flow} with $F=X^c_\phi$ ($\phi$ holomorphic on $\Omega$), $Y=DX^t(z)$ satisfies $Y^\top\Omega_c^{-1}Y=\Omega_c^{-1}$, $Y\Omega_cY^\top=\Omega_c$, $\det Y=\pm1$. For $\Phi=X^t$, $u,v,H$ holomorphic near $\Phi(z)$: (a) $\{u\circ\Phi,v\circ\Phi\}_c=\{u,v\}_c\circ\Phi$; (b) $D\Phi\,X^c_{H\circ\Phi}=X^c_H\circ\Phi$; (c) if $\dot y=X^c_{H\circ\Phi}(y)$ then $x=\Phi\circ y$ satisfies $\dot x=X^c_H(x)$.
\end{lemma}
\begin{proof}
$Y'=DF\,Y$, $Y(0)=1$ (Lemma~\ref{bpN:p5:flow}(b)); $DF=\Omega_cS$, $S$ the symmetric Hessian (\Kref{bp:p1:ext-schwarz}), $\Omega_c^\top=-\Omega_c$, so $DF^\top\Omega_c^{-1}=-S\Omega_c\Omega_c^{-1}=-S$ and $\Omega_c^{-1}DF=S$; $\frac d{dt}(Y^\top\Omega_c^{-1}Y)=Y^\top(-S+S)Y=0$. Inverting: $Y\Omega_cY^\top=\Omega_c$; determinants give $\det Y^2=1$. (a): $\nabla(u\circ\Phi)=Y^\top\nabla u\circ\Phi$ and $\{u,v\}_c=\nabla u^\top\Omega_c\nabla v$. (b), (c): chain rule and $Y\Omega_cY^\top=\Omega_c$.
\end{proof}

\begin{lemma}[Iterated brackets]\label{bpN:p11:adjoint}
Let $\phi\in\Os{R'}{s'}^e$, $u\in\Os Rs^e$ have finite norms, $0<\rho<R$, $0<\sigma<s$, $R+\rho\le R'$, $s+\sigma\le s'$, $\textup{(SL)}_{\rho,\sigma}$, $\eta=2\wn\phi{R'}{s'}/(\rho\sigma)$. Then $\operatorname{ad}^h_\phi u:=\{\operatorname{ad}^{h-1}_\phi u,\phi\}_c\in\Os{R-\rho}{s-\sigma}^e$ and for $h\ge1$
\[\frac1{h!}\wn{\operatorname{ad}^h_\phi u}{R-\rho}{s-\sigma}\le\Bigl(\frac{e^2\eta}8\Bigr)^h\wn uRs .\]
\end{lemma}
\begin{proof}
\proofstep{Setup}\label{stepN:p11:ad-setup} $\kappa=\rho/h$, $\lambda_h=\sigma/h$, $R_i=R-i\kappa$, $s_i=s-i\lambda_h$; $\textup{(SL)}_{\kappa,\lambda_h}$ holds (same ratio). By induction $\operatorname{ad}^i_\phi u\in\Os{R_i}{s_i}^e$.
\proofstep{One step}\label{stepN:p11:ad-step} Apply Lemma~\ref{bpN:p11:bracket} with $u:=\phi$ (radius $R'$, width $s'$), $v:=\operatorname{ad}^{i-1}_\phi u$ (radius $R_{i-1}$, width $s_{i-1}$) and losses $\kappa$, $\lambda_h$ (using $\{a,b\}_c=-\{b,a\}_c$). Since $R'-R_{i-1}+\kappa\ge(h+i)\kappa$ and $s'-s_{i-1}+\lambda_h\ge(h+i)\lambda_h$: $\wn{\operatorname{ad}^i_\phi u}{R_i}{s_i}\le\frac12\cdot\frac2{(h+i)\kappa\lambda_h}\wn\phi{}{}\wn{\operatorname{ad}^{i-1}_\phi u}{R_{i-1}}{s_{i-1}}$.
\proofstep{Product}\label{stepN:p11:ad-prod} $\prod_{i=1}^h(h+i)=(2h)!/h!$, $\kappa\lambda_h=\rho\sigma/h^2$: $\wn{\operatorname{ad}^h_\phi u}{}{}\le(h^2/(\rho\sigma))^h\frac{h!}{(2h)!}\wn\phi{}{}^h\wn u{}{}$, so $\frac1{h!}\wn{\operatorname{ad}^h_\phi u}{}{}\le\frac{(\eta h^2/2)^h}{(2h)!}\wn u{}{}\le\bigl(\frac{\eta h^2}2\cdot\frac{e^2}{4h^2}\bigr)^h\wn u{}{}$ using $(2h)!\ge(2h/e)^{2h}$ (\ref{bpN:p11:ext-num}).
\end{proof}

\begin{theorem}[Lemma B.5 for the extended class]\label{bpN:p11:composition}
Let $0<\rho<R\le R'-\rho$, $0<\sigma<s\le s'-\sigma$, $\phi\in\Os{R'}{s'}^e$ with finite norm, $\eta=2\wn\phi{R'}{s'}/(\rho\sigma)\le\frac12$, $\textup{(SL)}_{\rho,\sigma}$; $\Omega=\Dom{R'-\rho}{s'-\sigma}^e\supset\Dom Rs^e$, $\nu=\nu_{\rho,\sigma}$.
(i) $\nu(X^c_\phi)\le\eta/2\le\frac14$ on $\Omega$; the flow $X^t$, $\abs t\le1$, is defined and holomorphic on $\Omega^{1/2}\supset\Dom{R-\rho}{s-\sigma}^e$ and maps this set into $\Dom Rs^e$; precisely $X^t(z)\in\Dom{R-(1-\eta/2)\rho}{s-(1-\eta/2)\sigma}^e$.
(ii) For $u\in\Os Rs^e$ with finite norm, $u\circ X^1=\sum_{h\ge0}\frac1{h!}\operatorname{ad}^h_\phi u$ on $\Dom{R-\rho}{s-\sigma}^e$, the series converges in $\wn\cdot{R-\rho}{s-\sigma}$, and $\wn{u\circ X^1}{R-\rho}{s-\sigma}\le(1-\eta)^{-1}\wn uRs$.
(iii) The same for $X^t$ with $\eta\to\abs t\eta$. (iv) If $\phi,u$ are real on real points, so is $u\circ X^1$.
\end{theorem}
\begin{proof}
\proofstep{Field and domain}\label{stepN:p11:co-field} Lemma~\ref{bpN:p11:vf} (losses $\rho,\sigma$): $\nu(X^c_\phi)\le\wn\phi{}{}/(\rho\sigma)=\eta/2=:\varepsilon\le\frac14$ on $\Omega$. For $z\in\Dom{R-\rho}{s-\sigma}^e$, $\bar B_\nu(z,\frac12)\subset\Dom{R-\rho/2}{s-\sigma/2}^e\subset\Dom Rs^e\subset\Omega$ (Definition~\ref{bpN:p11:scaled}), so $z\in\Omega^{1/2}$; Lemma~\ref{bpN:p11:flow}(a): $X^t(z)\in\bar B_\nu(z,\varepsilon)\subset\Dom{R-(1-\varepsilon)\rho}{s-(1-\varepsilon)\sigma}^e$.
\proofstep{Series}\label{stepN:p11:co-series} Lemma~\ref{bpN:p11:adjoint}: $\frac1{h!}\operatorname{ad}^h_\phi u\in\Os{R-\rho}{s-\sigma}^e$ has norm $\le(e^2\eta/8)^h\wn uRs$; $e^2/8<0.93$ so $\sum_h\le(1-e^2\eta/8)^{-1}\le(1-\eta)^{-1}$. By B.1(a) the partial sums converge uniformly on $\Dom{R-\rho}{s-\sigma}^e$, the limit $S$ is holomorphic and periodic (\ref{bpN:p4:ext-weier}), its Fourier coefficients are the sums of the coefficients (termwise integration), and the norm bound follows by the triangle inequality.
\proofstep{Remainder}\label{stepN:p11:co-remainder} Let $z\in\Dom{R-\rho}{s-\sigma}^e$, $w=X^t(z)\in\Dom{R-\hat\rho}{s-\hat\sigma}^e$ with $\hat\rho=(1-\varepsilon)\rho$, $\hat\sigma=(1-\varepsilon)\sigma$ (ratio unchanged, so $\textup{(SL)}_{\hat\rho,\hat\sigma}$), $R+\hat\rho\le R'$. Lemma~\ref{bpN:p11:adjoint} with $(\hat\rho,\hat\sigma)$, $\hat\eta=\eta/(1-\varepsilon)^2\le\frac{16}9\eta\le\frac89$: $\wn{\operatorname{ad}^N_\phi u}{R-\hat\rho}{s-\hat\sigma}\le N!(e^2\hat\eta/8)^N\wn uRs$ and $e^2\hat\eta/8\le e^2/9<0.83$. The Taylor formula with integral remainder (Lemma~\ref{bpN:p5:taylor}, one real variable; derivatives $g^{(j)}=\operatorname{ad}^j_\phi u\circ X^t$ by Lemma~\ref{bpN:p11:symplectic} and $\frac d{dt}v(X^tz)=Dv\cdot X^c_\phi=\{v,\phi\}_c$) gives $\abs{R_N(z)}\le(e^2\hat\eta/8)^N\wn uRs\to0$, so $u(X^1z)=S(z)$. (iii) is (ii) for $t\phi$ ($X^t_\phi=X^1_{t\phi}$); (iv) by Lemma~\ref{bpN:p11:flow}(e).
\end{proof}

\begin{proposition}[Lie transform with first-order remainder; integral remainders]\label{bpN:p11:transform}
In the setting of Theorem~\ref{bpN:p11:composition} let $h\in\Os Rs^e$ (norm possibly infinite) with $w=\{h,\phi\}_c\in\Os Rs^e$ of finite norm, $f_0\in\Os Rs^e$ with finite norm. On $\Dom{R-\rho}{s-\sigma}^e$: $(h+f_0)\circ X^1=h+w+f_0+\mathcal R$, $\mathcal R=\int_0^1\{(1-t)w+f_0,\phi\}_c\circ X^t\,dt$, $\wn{\mathcal R}{R-\rho}{s-\sigma}\le\frac\eta{1-\eta}(\frac12\wn w{}{}+\wn{f_0}{}{})$. Moreover if $v_t\in\Os Rs^e$ ($0\le t\le1$) have $\sup_t\wn{v_t}Rs=V<\infty$ and $(t,z)\mapsto v_t(z),Dv_t(z)$ are continuous, then $G=\int_0^1v_t\circ X^t\,dt\in\Os{R-\rho}{s-\sigma}^e$ with $\wn G{R-\rho}{s-\sigma}\le(1-\eta)^{-1}V$.
\end{proposition}
\begin{proof}
\proofstep{Taylor}\label{stepN:p11:tr-taylor} Lemma~\ref{bpN:p5:taylor} (order $2$ for $h$, order $1$ for $f_0$) gives the identity pointwise (the trajectory stays in $\Dom Rs^e$ by Theorem~\ref{bpN:p11:composition}(i)). For the series form: $h\circ X^1=h+\sum_{i\ge0}\frac1{(i+1)!}\operatorname{ad}^i_\phi w$ (remainder $\le\frac1{N!}\max\abs{\operatorname{ad}^{N-1}_\phi w\circ X^t}\le(e^2\hat\eta/8)^{N-1}\wn w{}{}/N\to0$) and $f_0\circ X^1=\sum\frac1{j!}\operatorname{ad}^j_\phi f_0$; so $\mathcal R=\sum_{j\ge1}(\frac1{(j+1)!}\operatorname{ad}^j_\phi w+\frac1{j!}\operatorname{ad}^j_\phi f_0)$ and the bound is $\sum_{j\ge1}(e^2\eta/8)^j(\frac{\wn w{}{}}{j+1}+\wn{f_0}{}{})\le\frac\eta{1-\eta}(\frac12\wn w{}{}+\wn{f_0}{}{})$ ($e^2/8<1$).
\proofstep{Integral remainder}\label{stepN:p11:tr-int} $g_t=v_t\circ X^t\in\Os{R-\rho}{s-\sigma}^e$, $\wn{g_t}{}{}\le(1-\eta)^{-1}V$ by Theorem~\ref{bpN:p11:composition}(iii); $(t,z)\mapsto g_t(z)$ and $D_zg_t=Dv_t(X^tz)D_zX^t(z)$ are continuous; \ref{bpN:p4:ext-param} gives holomorphy of $G$, Fubini and Tonelli give $\wn GI{s-\sigma}\le\int_0^1\wn{g_t}I{s-\sigma}dt$.
\end{proof}

\begin{proposition}[Canonical map package]\label{bpN:p11:canonical-map}
Let $\phi\in\Os Rs^e$ have finite norm, $0<\rho<R/2$, $0<\sigma<s/2$, $\textup{(SL)}_{\rho,\sigma}$, $\varepsilon:=\wn\phi Rs/(\rho\sigma)\le\frac12$, $\Omega=\Dom{R-\rho}{s-\sigma}^e$, $\Phi=X^1_\phi$, $\Psi=X^{-1}_\phi$. Then: (a) $\Phi,\Psi$ are holomorphic on $\Dom{R-\frac32\rho}{s-\frac32\sigma}^e$ with values in $\Omega$, $\nu(\Phi-\mathrm{id})\le\varepsilon$, and $\abs{\Phi_\theta-\theta}\le\wn\phi{}{}/\rho$, $\abs{\Phi_I-I}\le\wn\phi{}{}/(e\sigma)$, $\abs{\Phi_{\tilde x}-\tilde x}\le c\wn\phi{}{}/\rho\le\frac1{16k}\wn\phi{}{}/\sigma$; (b) $\Phi(\theta+\ell,I,\tilde x)=\Phi(\theta,I,\tilde x)+(\ell,0,0)$, real points go to real points if $\phi$ is real; (c) $\abs{D\Phi-1}_\nu\le2\varepsilon$ on $\Dom{R-2\rho}{s-2\sigma}^e$; (d) $D\Phi\,\Omega_c\,D\Phi^\top=\Omega_c$; (e) $\Psi\circ\Phi=\mathrm{id}=\Phi\circ\Psi$ on $\Dom{R-2\rho}{s-2\sigma}^e$, and $\Phi$ maps it into $\Dom{R-\frac32\rho}{s-\frac32\sigma}^e$.
\end{proposition}
\begin{proof}
As Proposition~\ref{bpN:p5:canonical-map}, step by step: Lemma~\ref{bpN:p11:vf} gives $\nu(X^c_\phi)\le\varepsilon$ on the open set $\Omega$; $\bar B_\nu(z,c')\subset\Omega$ for $z\in\Dom{R-(1+c')\rho}{s-(1+c')\sigma}^e$ (Definition~\ref{bpN:p11:scaled}), so $\Dom{R-\frac32\rho}{\cdot}\subset\Omega^{1/2}$ and $\Dom{R-2\rho}{\cdot}\subset\Omega^1$; Lemma~\ref{bpN:p11:flow}(a)--(e) gives (a), (b), (c), (e) (the componentwise displacements are the mean value inequality along the trajectory with the bounds of Lemma~\ref{bpN:p11:vf}); (d) is Lemma~\ref{bpN:p11:symplectic}.
\end{proof}

\subsection{Iterative Lemma and Normal Form Lemma with slow parameters}\label{bpN:p11:s-nf}

Notation of Part~\ref{bpN:p6}: $(D,r,K,\alpha,L,\Lambda,h)\to(\{I_c\},R,K_a,\alpha,L,\Lambda,h)$, $K_a$ the analytic cut-off, $p>1$, $q=p/(p-1)$, $\ba=2\pi\alpha$, $\kappa=2\pi K_as$; $h$ is holomorphic on $V_R(I_c)$, regarded as a function on $\Dom Rs^e$ independent of $(\theta,\tilde x)$; $\omega=\partial_Ih$.

\begin{proposition}[Homological equation with parameters]\label{bpN:p11:hom}
Let $f\in\Os Rs^e$, $\delta=\wn fRs<\infty$, $\Lambda\subset\Z^n$, $K_a\ge0$, and assume (N-dir): $\abs{k\cdot\omega(I)}\ge\alpha/q$ on $V_R(I_c)$ for $k\in Z_{K_a}\setminus\Lambda$. Put $\xi=P_\Lambda T_{K_a}f$, $\phi=\sum_{k\in Z_{K_a}\setminus\Lambda}\frac{\hat f_k(I,\tilde x)}{2\pi ik\cdot\omega(I)}\chi_k(\theta)$. Then $\xi,\phi\in\Os Rs^e$, $\{\phi,h\}_c+\xi=T_{K_a}f$, $\wn\xi{}{},\wn{T_{K_a}f}{}{}\le\delta$, $\wn\phi{}{}\le q\delta/\ba$, $P_\Lambda\phi=0$; if $\Lambda=-\Lambda$ and $f$ is real then $\xi,\phi$ are real on real points.
\end{proposition}
\begin{proof}
Identical to Proposition~\ref{bpN:p6:homological} (with Corollary~\ref{bpN:p6:iterative-direct}): it is a statement about the Fourier coefficients in $\theta$ at fixed $\hat I=(I,\tilde x)$ (Lemma~\ref{bpN:p11:ext-part4}); $\hat\phi_k$ is holomorphic in $\hat I$ since $k\cdot\omega$ is holomorphic and bounded below; $\{\phi,h\}_c=\partial_\theta\phi\cdot\omega$ because $h$ depends on neither $\theta$ nor $\tilde x$ (the slow bracket of $h$ vanishes); the coefficients of $\partial_\theta\phi\cdot\omega$ are $2\pi i(k\cdot\omega)\hat\phi_k=\hat f_k$ for $k\in Z_{K_a}\setminus\Lambda$; reality as Step~\ref{stepN:p6:hom-real}. \qed
\end{proof}

\begin{theorem}[Iterative Lemma$^e$]\label{bpN:p11:iterative}
Let (N-dir) hold; $s,\rho,\sigma>0$, $2\rho<R$, $2\sigma<s$, $\textup{(SL)}_{\rho,\sigma}$; $H=h+g+f$ on $\Dom Rs^e$, $g=\sum_{j\in J}g_j$ as in Definition~\ref{bpN:p6:setting} (extended norms), $f\in\Os Rs^e$, $\delta=\wn fRs\le\ba\rho\sigma/(4q)$. Let $\phi$ be that of Proposition~\ref{bpN:p11:hom} and $\Phi=X^{1,c}_\phi$. Then:
(a) $\Phi$ is holomorphic on $\Dom{R-\frac32\rho}{s-\frac32\sigma}^e$ with values in $\Dom{R-\rho}{s-\sigma}^e$, $\theta$-translation equivariant, symplectic ($D\Phi\Omega_cD\Phi^\top=\Omega_c$), injective on $\Dom{R-2\rho}{s-2\sigma}^e$ (left inverse $X^{-1}_\phi$);
(b) $\abs{\Phi_\theta-\theta}\le q\delta/(\ba\rho)$, $\abs{\Phi_I-I}\le q\delta/(e\ba\sigma)$, $\abs{\Phi_{\tilde x}-\tilde x}\le\frac e{16k}\cdot\frac{q\delta}{e\ba\sigma}$;
(c) $H\circ\Phi=h+g_++f_+$ on $\Dom{R-2\rho}{s-2\sigma}^e$, $g_+=g+P_\Lambda T_{K_a}f$, $\wn{g_+-g}{}{}\le\delta$;
(d) $\wn{f_+}{R-2\rho}{s-2\sigma}\le\bigl(1-\frac{2q\delta}{\ba\rho\sigma}\bigr)^{-1}\bigl(\frac q{\ba\rho\sigma}(\abs g_*+\delta)+e^{-2\pi K_a\sigma}\bigr)\delta$;
(e) if $\Lambda=-\Lambda$ and $h,g,f$ are real on real points, so are $\phi,g_+,f_+$ and $\Phi$ preserves real points.
\end{theorem}
\begin{proof}
\proofstep{Generating function}\label{stepN:p11:it-gen} Proposition~\ref{bpN:p11:hom}: $\wn\phi{}{}\le q\delta/\ba$; $\eta:=2\wn\phi{}{}/(\rho\sigma)\le\frac{2q\delta}{\ba\rho\sigma}\le\frac12$.
\proofstep{(a), (b)}\label{stepN:p11:it-map} Proposition~\ref{bpN:p11:canonical-map} ($\rho<R/2$, $\sigma<s/2$, $\varepsilon\le\eta/2\le\frac14$).
\proofstep{Taylor identity}\label{stepN:p11:it-taylor} $f_0=g+T_{K_a}f$ (finite norm, $\wn g{}{}\le\sum\wn{g_j}{}{}$), $w=\{h,\phi\}_c=\xi-T_{K_a}f$. Proposition~\ref{bpN:p11:transform} with data $(R-\rho,s-\sigma;R,s)$ for $(u\text{-domain};\phi\text{-domain})$ (admissible: $2\rho<R$, $2\sigma<s$): on $\Dom{R-2\rho}{s-2\sigma}^e$, $(h+f_0)\circ\Phi=h+g+\xi+\mathcal R=h+g_++\mathcal R$, $\mathcal R=\int_0^1\{g+f_t,\phi\}_c\circ X^t\,dt$, $f_t=(1-t)\xi+tT_{K_a}f$. Then $H\circ\Phi=h+g_++f_+$, $f_+=\mathcal R+(f-T_{K_a}f)\circ\Phi$.
\proofstep{Estimates}\label{stepN:p11:it-est} $v_t=\{g+f_t,\phi\}_c$ is affine in $t$; Lemma~\ref{bpN:p11:bracket-g} (decomposition $g_j$ plus the term $f_t$, $\wn{f_t}{}{}\le\delta$, weight $1$): $\wn{v_t}{R-\rho}{s-\sigma}\le\frac1{\rho\sigma}(\abs g_*+\delta)\frac{q\delta}\ba=V$. Proposition~\ref{bpN:p11:transform}: $\wn{\mathcal R}{}{}\le(1-\eta)^{-1}V$. $\wn{f-T_{K_a}f}{R}{s-\sigma}\le e^{-2\pi K_a\sigma}\delta$ (Lemma~\ref{bpN:p4:projections}), and Theorem~\ref{bpN:p11:composition}(ii) gives $\wn{(f-T_{K_a}f)\circ\Phi}{}{}\le(1-\eta)^{-1}e^{-2\pi K_a\sigma}\delta$; $(1-\eta)^{-1}\le(1-2q\delta/(\ba\rho\sigma))^{-1}$. This is (d); (e) as Step~\ref{stepN:p6:it-real}.
\end{proof}

\begin{theorem}[Normal Form Lemma$^e$]\label{bpN:p11:nf}
Let $K_a>0$, $\Lambda=-\Lambda\subset\Z^n$, $(\{I_c\},R,K_a,\alpha,L,\Lambda,h)$ satisfy \ref{bpN:p6:cond-nr}, \ref{bpN:p6:cond-L}, $K_aLR\le\alpha/p$ (so (N-dir), Lemma~\ref{bpN:p6:divisor}); $R=r_B$ (the starting radius of the block), $s>0$, \textup{(SL)} ($c\le R/(32ks)$); $f\in\Os Rs^{e,\R}$ with $\wn fRs\le\varepsilon$ for a number $\varepsilon>0$, and
\[\textup{(i)}\ \ \varepsilon\le\frac{\alpha R}{2^7qK_a},\qquad\textup{(ii)}\ \ \kappa=2\pi K_as\ge6 .\]
Put $R_*=R/2$, $s_*=s/6$, $c_1=2qK_a/(\alpha R)$, $g_0=P_\Lambda T_{K_a}f$. There are a holomorphic injective $\Psi:\Dom{R_*}{s_*}^e\to\Dom Rs^e$ with $D\Psi\,\Omega_c\,D\Psi^\top=\Omega_c$, $\Psi(\theta+\ell,I,\tilde x)=\Psi(\theta,I,\tilde x)+(\ell,0,0)$, real at real points, and $g,f_*\in\Os{R_*}{s_*}^{e,\R}$ with $g$ a trigonometric polynomial in $\theta$ with frequencies in $\Lambda\cap Z_{K_a}$, such that $H\circ\Psi=h+g+f_*$ and
(a) $\wn{g-g_0}{R_*}{s_*}\le16c_1\varepsilon^2$; (b) $\wn{f_*}{R_*}{s_*}\le e^{-\kappa/6}\varepsilon$; (c) $\abs{\Pi_I\Psi-I}\le qK_a\varepsilon/\alpha\le R/2^7$, $\abs{\Pi_\theta\Psi-\theta}\le10q\varepsilon/(2\pi\alpha R)$, $\abs{\Pi_{\tilde x}\Psi-\tilde x}\le\frac18qK_a\varepsilon/\alpha\le R/2^{10}$ (physical slow displacement $\le\lambda R/2^{10}=w/2^{10}$).
\source{Pos93 pp.~192--196 and Theorem~\ref{bpN:p6:normal-form}; the displacement of the slow variables (c) is new.}
\end{theorem}
\begin{proof}
Write $\ba=2\pi\alpha$; $\ba s=\alpha\kappa/K_a$, $K_a\ge3/(\pi s)$. Indexing as in Theorem~\ref{bpN:p6:normal-form}: $V_i=\Dom{R_i}{s_i}^e$, $f_0=f$, $g_{-1}=0$; step $i$ is Theorem~\ref{bpN:p11:iterative} with losses $(\rho_i,\sigma_i)$ producing $\Phi_i:V_{i+1}\to V_i$, $g_i=g_{i-1}+P_\Lambda T_{K_a}f_i$, $R_{i+1}=R_i-2\rho_i$, $s_{i+1}=s_i-2\sigma_i$, $\varepsilon_i=\wn{f_i}{V_i}{}$ ($\varepsilon_0\le\varepsilon$; we carry the upper bound $\varepsilon$).
\proofstep{(SL) at every step}\label{stepN:p11:nf-sl} The losses are $(\rho_0,\sigma_0)=(\frac R8,\frac s6)$, $(\rho',\sigma')=(\frac R4,\frac s3)$, $(\rho,\sigma)=(\frac R{8N},\frac s{4N})$; $\sigma/\rho\in\{\frac{4s}{3R},\frac{4s}{3R},\frac{2s}R\}$, so $\sigma\le2s\rho/R$ and \textup{(SL)} gives $16kc\sigma\le\frac{R\sigma}{2s}\le\rho$.
\proofstep{Step 0: hypotheses}\label{stepN:p11:nf-first} $2\rho_0<R$, $2\sigma_0<s$, $\varepsilon\le\frac{\alpha R}{2^7qK_a}\le\frac{\alpha R}{2^7q}\frac{\pi s}3=\frac{\ba\rho_0\sigma_0}{16q}$ (as $\rho_0\sigma_0=Rs/48$), so $y_0:=\frac{q\varepsilon}{\ba\rho_0\sigma_0}\le\frac1{16}$, $u\le\frac14$.
\proofstep{Step 0: two cases}\label{stepN:p11:nf-case} Theorem~\ref{bpN:p11:iterative}(d) with $g=0$: $\wn{f_1}{}{}\le(1-2y_0)^{-1}(y_0+e^{-\kappa/6})\varepsilon$, $(1-2y_0)^{-1}\le\frac87$.
\emph{Case A, $e^{-\kappa/6}\le y_0$:} $\wn{f_1}{}{}\le\frac87\cdot2y_0\varepsilon=\frac{16}7y_0\varepsilon$, so $\varepsilon_1\le\frac{16}7\cdot\frac{48q\varepsilon^2}{\ba Rs}\le\frac{2^7q\varepsilon^2}{\ba Rs}\le\frac{2^7q\varepsilon}{\ba Rs}\cdot\frac{\alpha R}{2^7qK_a}\varepsilon=\frac\varepsilon\kappa\le\frac\varepsilon4$ (6.1).
\emph{Case B, $e^{-\kappa/6}>y_0$:} repeat with losses $(\frac R4,\frac s3)$ (then $u'=y_0/4$): $\wn{f_1}{R/2}{s/3}\le(1-\frac{y_0}2)^{-1}(\frac{y_0}4+e^{-2\kappa/6})\varepsilon\le\frac{32}{31}(\frac14+\frac1e)e^{-\kappa/6}\varepsilon<0.64\,e^{-\kappa/6}\varepsilon$ (using $y_0<e^{-\kappa/6}$, $e^{-2\kappa/6}\le e^{-1}e^{-\kappa/6}$). Stop: $\Psi=\Phi_0'$ on $\Dom{R/2}{s/3}^e\supset\Dom{R_*}{s_*}^e$, $g=g_0$, $f_*=f_1$; (a), (b) hold; displacement in Step~\ref{stepN:p11:nf-disp}. From now on Case A.
\proofstep{Number of steps}\label{stepN:p11:nf-N} $N=\lfloor\kappa/(12\ln2)\rfloor$; $\kappa\ge12\ln2\,N>8N$. If $N=0$: $\Psi=\Phi_0$, $g=g_0$, $f_*=f_1$, $\wn{f_*}{}{}\le\varepsilon/4\le e^{-\kappa/6}\varepsilon$ since $\kappa<12\ln2$ gives $e^{-\kappa/6}>\frac14$. For $N\ge1$, $i=1..N$: $\rho=\frac R{8N}$, $\sigma=\frac s{4N}$, $R_i=\frac34R-2(i-1)\rho$, $s_i=\frac23s-2(i-1)\sigma$; $R_{N+1}=R_*$, $s_{N+1}=s_*$; $2\pi K_a\sigma=\frac\kappa{4N}\ge3\ln2$, $e^{-2\pi K_a\sigma}\le\frac18$; $\rho\sigma=Rs/(32N^2)$; from (i), $\kappa>8N$: $\varepsilon\le\frac{\ba Rs}{2^{10}qN}$, hence by (6.1) $\varepsilon_1\le\frac{2^7q}{\ba Rs}\bigl(\frac{\ba Rs}{2^{10}qN}\bigr)^2=\frac{\ba\rho\sigma}{2^8q}$ (6.2).
\proofstep{Induction}\label{stepN:p11:nf-induct} Claim: step $i$ ($1\le i\le N$) is admissible and $\varepsilon_{i+1}\le\varepsilon_i/4$; so $\varepsilon_i\le4^{1-i}\varepsilon_1$. Admissibility: $R_i\ge R/2>2\rho$, $s_i>2\sigma$, $\varepsilon_i\le\varepsilon_1\le\ba\rho\sigma/(2^8q)\le\ba\rho\sigma/(4q)$. $\abs{g_{i-1}}_*$ (decomposition $g_0$ on $V_0$ and $g_j-g_{j-1}$ on $V_j$): the weights are $\le\frac12(\frac13+\frac13)\le\frac12$ for $j\ge1$ and $\le\frac12(\frac{R/8N}{R/4}+\frac{s/4N}{s/3})=\frac5{8N}\le\frac1N$ for $g_0$; so $\abs{g_{i-1}}_*\le\frac12\cdot\frac43\varepsilon_1+\frac\varepsilon N\le\frac{\ba\rho\sigma}{2^8q}+\frac{\ba\rho\sigma}{2^5q}\le\frac{\ba\rho\sigma}{2^4q}$ (using $\varepsilon/N\le\ba Rs/(2^{10}qN^2)=2^5\ba\rho\sigma/(2^{10}q)$). Theorem~\ref{bpN:p11:iterative}(d): $\frac q{\ba\rho\sigma}(\abs{g_{i-1}}_*+\varepsilon_i)\le\frac1{2^4}+\frac1{2^8}$, $e^{-2\pi K_a\sigma}\le\frac18$, $(1-2^{-7})^{-1}=\frac{128}{127}$, so $\varepsilon_{i+1}\le\frac{128}{127}(\frac1{16}+\frac1{256}+\frac18)\varepsilon_i<0.1930\,\varepsilon_i<\frac14\varepsilon_i$.
\proofstep{Conclusion}\label{stepN:p11:nf-concl} $\Psi=\Phi_0\circ\dots\circ\Phi_N$, $g=g_N$, $f_*=f_{N+1}$ on $\Dom{R_*}{s_*}^e$; $H_{i+1}=H_i\circ\Phi_i$ gives $H\circ\Psi=h+g+f_*$; Lemma~\ref{bpN:p6:telescope} (subsets of $\C^{2n+2k}$, maps with $D\Phi_i\Omega_cD\Phi_i^\top=\Omega_c$) gives holomorphy, symplecticity, periodicity, injectivity. (b): $\varepsilon_{N+1}\le4^{-N}\varepsilon_1\le4^{-N-1}\varepsilon\le e^{-\kappa/6}\varepsilon$ ($N+1>\kappa/(12\ln2)$). (a): $\wn{g-g_0}{}{}\le\frac43\varepsilon_1\le\frac43\frac{2^7q\varepsilon^2}{\ba Rs}$ and $\ba s=\alpha\kappa/K_a$, $\kappa>8$: $\le32\frac{qK_a\varepsilon^2}{\alpha R}=16c_1\varepsilon^2$.
\proofstep{Displacement}\label{stepN:p11:nf-disp} Lemma~\ref{bpN:p6:telescope} with the seminorms $\abs{\Pi_I\cdot}$, $\abs{\Pi_\theta\cdot}$, $\abs{\Pi_{\tilde x}\cdot}$ and Theorem~\ref{bpN:p11:iterative}(b). Case A: $\abs{\Pi_I\Psi-I}\le\frac{q\varepsilon}{e\ba\sigma_0}+\sum_{i=1}^N\frac{q\varepsilon_i}{e\ba\sigma}\le\frac{6q\varepsilon}{e\ba s}+\frac q{e\ba\sigma}\frac43\varepsilon_1$, with $\frac1{\ba s}\le\frac{K_a}{6\alpha}$, $\frac1{\ba\sigma}\le\frac{4K_a}{12\ln2\,\alpha}$, $\varepsilon_1\le\varepsilon/4$: $\le\frac{qK_a\varepsilon}{e\alpha}(1+\frac4{12\ln2}\cdot\frac43\cdot\frac14)\le1.17\frac{qK_a\varepsilon}{e\alpha}\le\frac{qK_a\varepsilon}\alpha$. $\abs{\Pi_\theta\Psi-\theta}\le\frac{q\varepsilon}{\ba\rho_0}+\frac q{\ba\rho}\frac43\varepsilon_1\le\frac{10q\varepsilon}{\ba R}$ (as in Part~\ref{bpN:p6}). Slow: each step's slow displacement is $\le\frac e{16k}$ times its $I$-displacement bound, so $\abs{\Pi_{\tilde x}\Psi-\tilde x}\le\frac e{16}\cdot1.17\frac{qK_a\varepsilon}{e\alpha}<\frac18\frac{qK_a\varepsilon}\alpha$; by (i) $\le R/2^{10}$. Case B: half of these.
\proofstep{Reality}\label{stepN:p11:nf-real} Theorem~\ref{bpN:p11:iterative}(e) at each step.
\end{proof}
